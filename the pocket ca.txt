import { useState, useEffect, useCallback } from 'react';
import { View, Text, StyleSheet, ScrollView, TouchableOpacity, RefreshControl, ActivityIndicator, Modal, TextInput, Alert, Dimensions } from 'react-native';
import { useRouter } from 'expo-router';
import { LinearGradient } from 'expo-linear-gradient';
import { TrendingUp, TrendingDown, IndianRupee, Camera, ArrowUpRight, ArrowDownRight, Receipt, Users, Plus, BarChart3, MessageCircle, Globe, Mic, AlertTriangle, Package, MicOff, ChevronRight, Store, ChevronDown, Check, Sparkles, Zap, Lightbulb, Brain } from 'lucide-react-native';
import { useAuth } from '@/contexts/AuthContext';
import { useI18n } from '@/contexts/I18nContext';
import { supabase } from '@/lib/supabase';
import AsyncStorage from '@react-native-async-storage/async-storage';

const { width } = Dimensions.get('window');

interface Transaction {
  id: string;
  description: string;
  amount: number;
  gst: number;
  total: number;
  date: string;
  type: 'sale' | 'expense';
}

interface UdhaarEntry {
  id: string;
  customer_name: string;
  customer_phone: string | null;
  description: string | null;
  amount: number;
  paid_amount: number;
  due_date: string | null;
  status: string;
  date_created: string;
}

interface Product {
  id: string;
  name: string;
  category: string | null;
  quantity: number;
  min_quantity: number;
  unit: string;
  price: number | null;
}

interface Store {
  id: string;
  name: string;
  address: string | null;
  phone: string | null;
  gstin: string | null;
  is_default: boolean;
}

interface MonthlyData {
  month: string;
  sales: number;
  expenses: number;
  profit: number;
}

const MONTHS_EN = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
const MONTHS_HI = ['जन', 'फर', 'मार्च', 'अप्रै', 'मई', 'जून', 'जुला', 'अग', 'सित', 'अक्टू', 'नव', 'दिस'];

const SELECTED_STORE_KEY = '@pocket_ca_selected_store';

export default function DashboardScreen() {
  const { user } = useAuth();
  const { t, language, toggleLanguage } = useI18n();
  const router = useRouter();
  const [refreshing, setRefreshing] = useState(false);
  const [loading, setLoading] = useState(true);
  const [totalSales, setTotalSales] = useState(0);
  const [totalExpenses, setTotalExpenses] = useState(0);
  const [totalUdhaar, setTotalUdhaar] = useState(0);
  const [recentTransactions, setRecentTransactions] = useState<Transaction[]>([]);
  const [udhaarEntries, setUdhaarEntries] = useState<UdhaarEntry[]>([]);
  const [lowStockProducts, setLowStockProducts] = useState<Product[]>([]);
  const [monthlyData, setMonthlyData] = useState<MonthlyData[]>([]);
  const [stores, setStores] = useState<Store[]>([]);
  const [selectedStore, setSelectedStore] = useState<Store | null>(null);
  const [showStoreDropdown, setShowStoreDropdown] = useState(false);
  const [showStoreModal, setShowStoreModal] = useState(false);
  const [showUdhaarModal, setShowUdhaarModal] = useState(false);
  const [showProductModal, setShowProductModal] = useState(false);
  const [showVoiceModal, setShowVoiceModal] = useState(false);
  const [isListening, setIsListening] = useState(false);
  const [voiceTranscript, setVoiceTranscript] = useState('');
  const [udhaarForm, setUdhaarForm] = useState({ customer_name: '', customer_phone: '', description: '', amount: '' });
  const [productForm, setProductForm] = useState({ name: '', category: '', quantity: '', min_quantity: '5', unit: 'pcs', price: '' });
  const [storeForm, setStoreForm] = useState({ name: '', address: '', phone: '', gstin: '' });
  const [isOffline, setIsOffline] = useState(false);
  const [aiInsights, setAiInsights] = useState<string[]>([]);
  const [forecastData, setForecastData] = useState<{ month: string; predictedSales: number }[]>([]);

  const MONTHS = language === 'hi' ? MONTHS_HI : MONTHS_EN;

  // Check connectivity
  useEffect(() => {
    const checkConnectivity = () => {
      setIsOffline(!navigator.onLine);
    };

    window.addEventListener('online', () => setIsOffline(false));
    window.addEventListener('offline', () => setIsOffline(true));
    checkConnectivity();

    return () => {
      window.removeEventListener('online', () => setIsOffline(false));
      window.removeEventListener('offline', () => setIsOffline(true));
    };
  }, []);

  // Fetch stores
  const fetchStores = async () => {
    if (!user) return;
    try {
      const { data, error } = await supabase
        .from('stores')
        .select('*')
        .eq('user_id', user.id)
        .order('created_at', { ascending: true });

      if (error) throw error;

      const storeList = data || [];
      setStores(storeList);

      // Load saved store selection
      const savedStoreId = await AsyncStorage.getItem(SELECTED_STORE_KEY);
      if (savedStoreId) {
        const saved = storeList.find(s => s.id === savedStoreId);
        if (saved) setSelectedStore(saved);
      }

      // If no store selected and stores exist, select first or default
      if (!selectedStore && storeList.length > 0) {
        const defaultStore = storeList.find(s => s.is_default) || storeList[0];
        setSelectedStore(defaultStore);
        await AsyncStorage.setItem(SELECTED_STORE_KEY, defaultStore.id);
      }

      // Create default store if none exist
      if (storeList.length === 0) {
        const { data: newStore } = await supabase
          .from('stores')
          .insert({
            name: language === 'hi' ? 'मेरी दुकान' : 'My Shop',
            user_id: user.id,
            is_default: true,
          })
          .select()
          .single();

        if (newStore) {
          setStores([newStore]);
          setSelectedStore(newStore);
          await AsyncStorage.setItem(SELECTED_STORE_KEY, newStore.id);
        }
      }
    } catch (err) {
      console.error('Error fetching stores:', err);
    }
  };

  const fetchData = async () => {
    if (!user) return;

    try {
      const currentYear = new Date().getFullYear();
      const startOfYear = `${currentYear}-01-01`;
      const storeId = selectedStore?.id;

      const query = storeId
        ? { eq: { store_id: storeId }, also: { user_id: user.id } }
        : { eq: { user_id: user.id } };

      const [salesRes, expensesRes, udhaarRes, productsRes, allSalesRes, allExpensesRes] = await Promise.all([
        storeId
          ? supabase.from('sales').select('*').eq('store_id', storeId).order('date', { ascending: false }).limit(5)
          : supabase.from('sales').select('*').eq('user_id', user.id).order('date', { ascending: false }).limit(5),
        storeId
          ? supabase.from('expenses').select('*').eq('store_id', storeId).order('date', { ascending: false }).limit(5)
          : supabase.from('expenses').select('*').eq('user_id', user.id).order('date', { ascending: false }).limit(5),
        storeId
          ? supabase.from('udhaar_entries').select('*').eq('store_id', storeId).in('status', ['pending', 'partial']).order('date_created', { ascending: false }).limit(5)
          : supabase.from('udhaar_entries').select('*').eq('user_id', user.id).in('status', ['pending', 'partial']).order('date_created', { ascending: false }).limit(5),
        storeId
          ? supabase.from('products').select('*').eq('store_id', storeId)
          : supabase.from('products').select('*').eq('user_id', user.id),
        storeId
          ? supabase.from('sales').select('total, date').eq('store_id', storeId).gte('date', startOfYear)
          : supabase.from('sales').select('total, date').eq('user_id', user.id).gte('date', startOfYear),
        storeId
          ? supabase.from('expenses').select('total, date').eq('store_id', storeId).gte('date', startOfYear)
          : supabase.from('expenses').select('total, date').eq('user_id', user.id).gte('date', startOfYear),
      ]);

      const sales = salesRes.data || [];
      const expenses = expensesRes.data || [];
      const udhaar = udhaarRes.data || [];
      const products = productsRes.data || [];
      const allSales = allSalesRes.data || [];
      const allExpenses = allExpensesRes.data || [];

      const salesTotal = sales.reduce((sum, s) => sum + Number(s.total), 0);
      const expensesTotal = expenses.reduce((sum, e) => sum + Number(e.total), 0);
      const udhaarTotal = udhaar.reduce((sum, u) => sum + Number(u.amount) - Number(u.paid_amount), 0);

      setTotalSales(salesTotal);
      setTotalExpenses(expensesTotal);
      setTotalUdhaar(udhaarTotal);

      const months: MonthlyData[] = MONTHS.map((month) => ({ month, sales: 0, expenses: 0, profit: 0 }));

      (allSales || []).forEach((sale) => {
        const monthIndex = new Date(sale.date).getMonth();
        months[monthIndex].sales += Number(sale.total);
      });

      (allExpenses || []).forEach((expense) => {
        const monthIndex = new Date(expense.date).getMonth();
        months[monthIndex].expenses += Number(expense.total);
      });

      months.forEach((m) => { m.profit = m.sales - m.expenses; });

      const currentMonth = new Date().getMonth();
      setMonthlyData(months.slice(0, currentMonth + 1));

      const salesTransactions: Transaction[] = sales.map(s => ({
        id: s.id, description: s.description, amount: Number(s.amount), gst: Number(s.gst), total: Number(s.total), date: s.date, type: 'sale' as const,
      }));

      const expenseTransactions: Transaction[] = expenses.map(e => ({
        id: e.id, description: e.description, amount: Number(e.amount), gst: Number(e.gst), total: Number(e.total), date: e.date, type: 'expense' as const,
      }));

      const combined = [...salesTransactions, ...expenseTransactions].sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()).slice(0, 6);
      setRecentTransactions(combined);
      setUdhaarEntries(udhaar);
      setLowStockProducts(products.filter((p: Product) => p.quantity <= p.min_quantity));
    } catch (err) {
      console.error('Error:', err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    const init = async () => {
      setLoading(true);
      await fetchStores();
    };
    init();
  }, [user]);

  useEffect(() => {
    if (selectedStore) fetchData();
  }, [selectedStore]);

  const onRefresh = useCallback(async () => { setRefreshing(true); await fetchData(); setRefreshing(false); }, [selectedStore, user]);

  const selectStore = async (store: Store) => {
    setSelectedStore(store);
    await AsyncStorage.setItem(SELECTED_STORE_KEY, store.id);
    setShowStoreDropdown(false);
  };

  const addStore = async () => {
    if (!user || !storeForm.name) { Alert.alert(t('pleaseFillFields')); return; }
    try {
      const { data, error } = await supabase
        .from('stores')
        .insert({
          name: storeForm.name,
          address: storeForm.address || null,
          phone: storeForm.phone || null,
          gstin: storeForm.gstin || null,
          user_id: user.id,
          is_default: stores.length === 0,
        })
        .select()
        .single();

      if (error) throw error;
      if (data) {
        setStores([...stores, data]);
        setSelectedStore(data);
        await AsyncStorage.setItem(SELECTED_STORE_KEY, data.id);
      }
      setShowStoreModal(false);
      setStoreForm({ name: '', address: '', phone: '', gstin: '' });
    } catch (err) { console.error('Error adding store:', err); }
  };

  const formatCurrency = (amount: number) => new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR' }).format(amount);
  const formatShortCurrency = (amount: number) => {
    if (amount >= 100000) return `${(amount / 100000).toFixed(1)}L`;
    if (amount >= 1000) return `${(amount / 1000).toFixed(1)}K`;
    return amount.toString();
  };
  const formatShortDate = (dateStr: string) => new Date(dateStr).toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });

  const sendWhatsAppReminder = (entry: UdhaarEntry) => {
    if (!entry.customer_phone) { Alert.alert('Error', 'No phone number available'); return; }
    const phone = entry.customer_phone.replace(/\D/g, '');
    const pendingAmount = Number(entry.amount) - Number(entry.paid_amount);
    const message = language === 'hi'
      ? `नमस्ते ${entry.customer_name}, आपका ${formatCurrency(pendingAmount)} बकाया है। कृपया जल्द भुगतान करें। - The Pocket CA`
      : `Hello ${entry.customer_name}, you have a pending payment of ${formatCurrency(pendingAmount)}. Please pay soon. - The Pocket CA`;
    window.open(`https://wa.me/${phone}?text=${encodeURIComponent(message)}`, '_blank');
  };

  const addUdhaar = async () => {
    if (!user || !udhaarForm.customer_name || !udhaarForm.amount) { Alert.alert(t('pleaseFillFields')); return; }
    try {
      await supabase.from('udhaar_entries').insert({
        customer_name: udhaarForm.customer_name, customer_phone: udhaarForm.customer_phone,
        description: udhaarForm.description, amount: parseFloat(udhaarForm.amount), paid_amount: 0, status: 'pending',
        date_created: new Date().toISOString().split('T')[0], user_id: user.id, store_id: selectedStore?.id,
      });
      setShowUdhaarModal(false);
      setUdhaarForm({ customer_name: '', customer_phone: '', description: '', amount: '' });
      fetchData();
    } catch (err) { console.error('Error adding udhaar:', err); }
  };

  const addProduct = async () => {
    if (!user || !productForm.name || !productForm.quantity) { Alert.alert(t('pleaseFillFields')); return; }
    try {
      await supabase.from('products').insert({
        name: productForm.name, category: productForm.category, quantity: parseInt(productForm.quantity),
        min_quantity: parseInt(productForm.min_quantity) || 5, unit: productForm.unit, price: productForm.price ? parseFloat(productForm.price) : null,
        user_id: user.id, store_id: selectedStore?.id,
      });
      setShowProductModal(false);
      setProductForm({ name: '', category: '', quantity: '', min_quantity: '5', unit: 'pcs', price: '' });
      fetchData();
    } catch (err) { console.error('Error adding product:', err); }
  };

  const startVoiceEntry = () => {
    setShowVoiceModal(true);
    setVoiceTranscript('');
    setIsListening(true);
    setTimeout(() => {
      setIsListening(false);
      setVoiceTranscript(language === 'hi' ? 'बिक्री 500 रुपये' : 'Add sale 500 rupees');
    }, 2000);
  };

  const processVoiceCommand = async () => {
    const amountMatch = voiceTranscript.match(/(\d+)/);
    if (!amountMatch) { Alert.alert('Error', 'Could not understand amount'); return; }
    const amount = parseFloat(amountMatch[1]);
    const isSale = voiceTranscript.toLowerCase().includes('sale') || voiceTranscript.toLowerCase().includes('बिक्री');

    if (!user) return;

    try {
      if (isSale) {
        await supabase.from('sales').insert({
          description: language === 'hi' ? 'सेवा प्रविष्टि' : 'Voice Entry',
          amount: amount, gst: 0, total: amount,
          date: new Date().toISOString().split('T')[0], user_id: user.id, store_id: selectedStore?.id,
        });
      } else {
        await supabase.from('expenses').insert({
          description: language === 'hi' ? 'वॉइस प्रविष्टि' : 'Voice Entry',
          amount: amount, gst: 0, total: amount,
          date: new Date().toISOString().split('T')[0], user_id: user.id, store_id: selectedStore?.id,
        });
      }
      setShowVoiceModal(false);
      fetchData();
    } catch (err) { console.error('Error:', err); }
  };

  const profit = totalSales - totalExpenses;
  const maxProfit = Math.max(...monthlyData.map(m => Math.abs(m.profit)), 1);

  // Simulated AI Forecast Function
  const generateForecast = useCallback(() => {
    if (monthlyData.length < 3) return [];

    const last3Months = monthlyData.slice(-3);
    const avgGrowth = last3Months.reduce((sum, m, i) => {
      if (i === 0) return 0;
      const prevSales = last3Months[i - 1].sales;
      if (prevSales === 0) return sum;
      return sum + ((m.sales - prevSales) / prevSales) * 100;
    }, 0) / 2;

    const lastMonth = monthlyData[monthlyData.length - 1];
    const currentMonthIndex = new Date().getMonth();

    // Simulate next 2 months forecast
    const nextMonth1Sales = lastMonth.sales * (1 + avgGrowth / 100 + (Math.random() * 0.1 - 0.05));
    const nextMonth2Sales = nextMonth1Sales * (1 + avgGrowth / 100 + (Math.random() * 0.1 - 0.05));

    return [
      { month: MONTHS[(currentMonthIndex + 1) % 12], predictedSales: Math.max(0, nextMonth1Sales) },
      { month: MONTHS[(currentMonthIndex + 2) % 12], predictedSales: Math.max(0, nextMonth2Sales) },
    ];
  }, [monthlyData, MONTHS]);

  // Generate AI Insights
  const generateAIInsights = useCallback(() => {
    const insights: string[] = [];
    const language_en = language === 'en';

    if (monthlyData.length >= 2) {
      const lastMonth = monthlyData[monthlyData.length - 1];
      const prevMonth = monthlyData[monthlyData.length - 2];

      // Trend Analysis
      const salesGrowth = prevMonth.sales > 0 ? ((lastMonth.sales - prevMonth.sales) / prevMonth.sales) * 100 : 0;
      if (salesGrowth > 10) {
        insights.push(language_en
          ? `Revenue trending up ${salesGrowth.toFixed(0)}% this month - consider increasing inventory`
          : `इस महीने राजस्व ${salesGrowth.toFixed(0)}% बढ़ा - इन्वेंटरी बढ़ाएं`);
      } else if (salesGrowth < -10) {
        insights.push(language_en
          ? `Sales down ${Math.abs(salesGrowth).toFixed(0)}% - consider promotional offers`
          : `बिक्री ${Math.abs(salesGrowth).toFixed(0)}% कम - प्रमोशनल ऑफर दें`);
      }

      // Profit Margin Analysis
      const profitMargin = lastMonth.sales > 0 ? (lastMonth.profit / lastMonth.sales) * 100 : 0;
      if (profitMargin < 10 && lastMonth.sales > 0) {
        insights.push(language_en
          ? `Profit margin at ${profitMargin.toFixed(0)}% - review pricing strategy`
          : `लाभ मार्जिन ${profitMargin.toFixed(0)}% - मूल्य नीति की समीक्षा करें`);
      }

      // Seasonal Insight
      const currentMonth = new Date().getMonth();
      if (currentMonth >= 9 || currentMonth <= 0) {
        insights.push(language_en
          ? 'Festive season ahead - stock up on popular items for increased demand'
          : 'त्योहारों का मौसम - लोकप्रिय वस्तुओं का स्टॉक बढ़ाएं');
      } else if (currentMonth >= 3 && currentMonth <= 5) {
        insights.push(language_en
          ? 'Summer approaching - beverages and cooling products may see higher demand'
          : 'गर्मी आ रही है - पेय पदार्थों की मांग बढ़ सकती है');
      } else if (currentMonth >= 6 && currentMonth <= 8) {
        insights.push(language_en
          ? 'Monsoon season - umbrella and rain gear demand expected to rise'
          : 'मानसून - छाता और बारिश का सामान बेचने का सही समय');
      }
    }

    // Low Stock Insight
    if (lowStockProducts.length > 0) {
      insights.push(language_en
        ? `${lowStockProducts.length} items running low - reorder to avoid stockouts`
        : `${lowStockProducts.length} वस्तुएं कम हैं - स्टॉकआउट से बचने के लिए पुनः ऑर्डर करें`);
    }

    // Udhaar Insight
    if (totalUdhaar > totalSales * 0.2 && totalSales > 0) {
      insights.push(language_en
        ? 'Pending dues exceed 20% of sales - follow up with customers'
        : 'बकाया बिक्री का 20% से अधिक - ग्राहकों को रिमाइंडर भेजें');
    }

    // Expense Analysis
    if (totalExpenses > totalSales * 0.7 && totalSales > 0) {
      insights.push(language_en
        ? 'Expenses high relative to revenue - look for cost optimization opportunities'
        : 'खर्च अधिक है - लागत अनुकूलन के अवसर तलाशें');
    }

    // Default insight if none generated
    if (insights.length === 0) {
      insights.push(language_en
        ? 'Business metrics look healthy - continue current strategies'
        : 'व्यापार मेट्रिक्स अच्छे हैं - वर्तमान रणनीति जारी रखें');
    }

    return insights.slice(0, 4);
  }, [monthlyData, lowStockProducts, totalUdhaar, totalSales, totalExpenses, language]);

  useEffect(() => {
    if (monthlyData.length > 0) {
      setForecastData(generateForecast());
      setAiInsights(generateAIInsights());
    }
  }, [monthlyData, lowStockProducts, totalUdhaar, generateForecast, generateAIInsights]);

  if (loading) {
    return (
      <View style={styles.container}>
        <View style={styles.loadingContainer}><ActivityIndicator size="large" color="#10b981" /></View>
      </View>
    );
  }

  return (
    <View style={styles.container}>

                {selectedStore?.id === store.id && <Check color="#10b981" size={18} />}

            ))}
            <TouchableOpacity style={styles.addStoreBtn} onPress={() => { setShowStoreDropdown(false); setShowStoreModal(true); }}>
              <Plus color="#10b981" size={18} />

        {/* Summary Cards */}
        <View style={styles.statsContainer}>
          <View style={[styles.statCard, styles.salesCard]}>
            <View style={[styles.statIconContainer, { backgroundColor: 'rgba(16, 185, 129, 0.2)' }]}>
              <TrendingUp color="#10b981" size={22} strokeWidth={2} />
            </View>
            <Text style={styles.statLabel}>{t('totalSales')}</Text>
            <Text style={[styles.statValue, { color: '#10b981' }]}>{formatCurrency(totalSales)}</Text>
          </View>
          <View style={[styles.statCard, styles.expenseCard]}>
            <View style={[styles.statIconContainer, { backgroundColor: 'rgba(239, 68, 68, 0.2)' }]}>
              <TrendingDown color="#ef4444" size={22} strokeWidth={2} />
            </View>
            <Text style={styles.statLabel}>{t('totalExpenses')}</Text>
            <Text style={[styles.statValue, { color: '#ef4444' }]}>{formatCurrency(totalExpenses)}</Text>
          </View>
        </View>

        {/* Profit Card */}
        <TouchableOpacity style={styles.profitCard} onPress={() => router.push('/(tabs)/reports')}>
          <View style={styles.profitHeader}>
            <View style={styles.profitHeaderLeft}>
              <IndianRupee color="#f8fafc" size={20} strokeWidth={2} />

                <View style={[styles.legendDot, { backgroundColor: '#00f5ff' }]} />

                <View style={[styles.legendDot, { backgroundColor: '#ff4d6d' }]} />

            <TextInput style={styles.modalInput} placeholder="GSTIN" placeholderTextColor="#64748b" value={storeForm.gstin} onChangeText={(text) => setStoreForm({ ...storeForm, gstin: text })} />
            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.modalCancelBtn} onPress={() => setShowStoreModal(false)}><Text style={styles.modalCancelText}>{t('cancel')}</Text></TouchableOpacity>
              <TouchableOpacity style={styles.modalSaveBtn} onPress={addStore}><Text style={styles.modalSaveText}>{t('save')}</Text></TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* Product Modal */}
      <Modal visible={showProductModal} transparent animationType="slide" onRequestClose={() => setShowProductModal(false)}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>{t('addProduct')}</Text>
            <TextInput style={styles.modalInput} placeholder={t('productName')} placeholderTextColor="#64748b" value={productForm.name} onChangeText={(text) => setProductForm({ ...productForm, name: text })} />
            <TextInput style={styles.modalInput} placeholder={t('category')} placeholderTextColor="#64748b" value={productForm.category} onChangeText={(text) => setProductForm({ ...productForm, category: text })} />
            <View style={styles.modalRow}>
              <TextInput style={[styles.modalInput, styles.modalInputHalf]} placeholder={t('quantity')} placeholderTextColor="#64748b" keyboardType="numeric" value={productForm.quantity} onChangeText={(text) => setProductForm({ ...productForm, quantity: text })} />
              <TextInput style={[styles.modalInput, styles.modalInputHalf]} placeholder={t('minQuantity')} placeholderTextColor="#64748b" keyboardType="numeric" value={productForm.min_quantity} onChangeText={(text) => setProductForm({ ...productForm, min_quantity: text })} />
            </View>
            <TextInput style={styles.modalInput} placeholder={t('unit')} placeholderTextColor="#64748b" value={productForm.unit} onChangeText={(text) => setProductForm({ ...productForm, unit: text })} />
            <TextInput style={styles.modalInput} placeholder={t('price')} placeholderTextColor="#64748b" keyboardType="decimal-pad" value={productForm.price} onChangeText={(text) => setProductForm({ ...productForm, price: text })} />
            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.modalCancelBtn} onPress={() => setShowProductModal(false)}><Text style={styles.modalCancelText}>{t('cancel')}</Text></TouchableOpacity>
              <TouchableOpacity style={styles.modalSaveBtn} onPress={addProduct}><Text style={styles.modalSaveText}>{t('save')}</Text></TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* Voice Modal */}
      <Modal visible={showVoiceModal} transparent animationType="slide" onRequestClose={() => setShowVoiceModal(false)}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>{t('voiceEntry')}</Text>
            <View style={styles.voiceContainer}>
              {isListening ? (
                <>
                  <View style={styles.voiceIconPulse}>
                    <Mic color="#10b981" size={48} strokeWidth={2} />
                  </View>
                  <Text style={styles.voiceStatus}>{t('listening')}</Text>
                </>
              ) : (
                <>
                  <MicOff color="#64748b" size={48} strokeWidth={2} />
                  <Text style={styles.voiceTranscript}>{voiceTranscript}</Text>
                </>
              )}
            </View>
            <Text style={styles.voiceHint}>{t('voiceHint')}</Text>
            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.modalCancelBtn} onPress={() => setShowVoiceModal(false)}><Text style={styles.modalCancelText}>{t('cancel')}</Text></TouchableOpacity>
              {!isListening && voiceTranscript && (
                <TouchableOpacity style={styles.modalSaveBtn} onPress={processVoiceCommand}><Text style={styles.modalSaveText}>{t('save')}</Text></TouchableOpacity>
              )}
            </View>
            <Text style={styles.voiceWebNote}>{t('voiceNotSupported')}</Text>
          </View>
        </View>
      </Modal>
    </View>
  );
}

  import { useState, useEffect, useCallback } from 'react';
  import { View, Text, StyleSheet, ScrollView, TouchableOpacity, RefreshControl, ActivityIndicator, Modal, TextInput, Alert } from 'react-native';
  import { useRouter } from 'expo-router';
  import { Users, Plus, MessageCircle, Phone, IndianRupee, ChevronRight, Globe, Trash2, CheckCircle } from 'lucide-react-native';
  import { useAuth } from '@/contexts/AuthContext';
  import { useI18n } from '@/contexts/I18nContext';
  import { supabase } from '@/lib/supabase';
  
  interface Customer {
    id: string;
    customer_name: string;
    customer_phone: string | null;
    description: string | null;
    amount: number;
    paid_amount: number;
    due_date: string | null;
    status: string;
    date_created: string;
  }
  
  export default function KhataScreen() {
    const { user } = useAuth();
    const { t, language, toggleLanguage } = useI18n();
    const router = useRouter();
    const [refreshing, setRefreshing] = useState(false);
    const [loading, setLoading] = useState(true);
    const [customers, setCustomers] = useState<Customer[]>([]);
    const [totalUdhaar, setTotalUdhaar] = useState(0);
    const [showAddModal, setShowAddModal] = useState(false);
    const [selectedCustomer, setSelectedCustomer] = useState<Customer | null>(null);
    const [showPaymentModal, setShowPaymentModal] = useState(false);
    const [paymentAmount, setPaymentAmount] = useState('');
    const [form, setForm] = useState({ customer_name: '', customer_phone: '', amount: '', description: '', due_date: '' });
  
    const fetchData = async () => {
      if (!user) return;
  
      try {
        const { data, error } = await supabase
          .from('udhaar_entries')
          .select('*')
          .eq('user_id', user.id)
          .order('date_created', { ascending: false });
  
        if (error) throw error;
  
        const entries = data || [];
        setCustomers(entries);
        const total = entries.reduce((sum, u) => sum + Number(u.amount) - Number(u.paid_amount), 0);
        setTotalUdhaar(total);
      } catch (err) {
        console.error('Error:', err);
      } finally {
        setLoading(false);
      }
    };
  
    useEffect(() => { fetchData(); }, [user]);
  
    const onRefresh = useCallback(async () => { setRefreshing(true); await fetchData(); setRefreshing(false); }, [user]);
  
    const formatCurrency = (amount: number) => new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR' }).format(amount);
    const formatDate = (dateStr: string) => new Date(dateStr).toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' });
  
    const sendWhatsAppReminder = (customer: Customer) => {
      if (!customer.customer_phone) { Alert.alert('Error', language === 'hi' ? 'फोन नंबर नहीं है' : 'No phone number available'); return; }
      const phone = customer.customer_phone.replace(/\D/g, '');
      const pendingAmount = Number(customer.amount) - Number(customer.paid_amount);
      const message = language === 'hi'
        ? `नमस्ते ${customer.customer_name}, आपका ${formatCurrency(pendingAmount)} बकाया है। कृपया जल्द भुगतान करें। - The Pocket CA`
        : `Hello ${customer.customer_name}, you have a pending payment of ${formatCurrency(pendingAmount)}. Please pay soon. - The Pocket CA`;
      window.open(`https://wa.me/${phone}?text=${encodeURIComponent(message)}`, '_blank');
    };
  
    const addCustomer = async () => {
      if (!user || !form.customer_name || !form.amount) { Alert.alert(t('pleaseFillFields')); return; }
      try {
        await supabase.from('udhaar_entries').insert({
          customer_name: form.customer_name, customer_phone: form.customer_phone,
          description: form.description, amount: parseFloat(form.amount), paid_amount: 0, status: 'pending',
          due_date: form.due_date || null, date_created: new Date().toISOString().split('T')[0], user_id: user.id,
        });
        setShowAddModal(false);
        setForm({ customer_name: '', customer_phone: '', amount: '', description: '', due_date: '' });
        fetchData();
      } catch (err) { console.error('Error adding customer:', err); }
    };
  
    const recordPayment = async () => {
      if (!selectedCustomer || !paymentAmount) return;
      const amount = parseFloat(paymentAmount);
      const newPaidAmount = Number(selectedCustomer.paid_amount) + amount;
      const newStatus = newPaidAmount >= Number(selectedCustomer.amount) ? 'paid' : 'partial';
  
      try {
        await supabase
          .from('udhaar_entries')
          .update({ paid_amount: newPaidAmount, status: newStatus })
          .eq('id', selectedCustomer.id);
        setShowPaymentModal(false);
        setPaymentAmount('');
        setSelectedCustomer(null);
        fetchData();
      } catch (err) { console.error('Error:', err); }
    };
  
    const deleteEntry = async (id: string) => {
      try {
        await supabase.from('udhaar_entries').delete().eq('id', id);
        fetchData();
      } catch (err) { console.error('Error deleting:', err); }
    };
  
    const pendingCustomers = customers.filter(c => c.status !== 'paid');
    const paidCustomers = customers.filter(c => c.status === 'paid');
  
    if (loading) {
      return (
        <View style={styles.container}>
          <View style={styles.loadingContainer}><ActivityIndicator size="large" color="#10b981" /></View>
        </View>
      );
    }
  
    return (
      <View style={styles.container}>
        <View style={styles.header}>
          <View>
            <Text style={styles.title}>{t('udhaarKhata')}</Text>
            <Text style={styles.subtitle}>{t('customerName')}</Text>
          </View>
          <TouchableOpacity style={styles.languageBtn} onPress={toggleLanguage}>
            <Globe color="#94a3b8" size={18} />
            <Text style={styles.languageText}>{language === 'en' ? 'हि' : 'EN'}</Text>
          </TouchableOpacity>
        </View>
  
        {/* Summary Card */}
        <View style={styles.summaryCard}>
          <View style={styles.summaryHeader}>
            <Users color="#f59e0b" size={24} strokeWidth={2} />
            <Text style={styles.summaryTitle}>{t('totalUdhaar')}</Text>
          </View>
          <Text style={styles.summaryAmount}>{formatCurrency(totalUdhaar)}</Text>
          <Text style={styles.summaryCount}>{pendingCustomers.length} {language === 'hi' ? 'ग्राहक बकाया' : 'customers pending'}</Text>
        </View>
  
        <TouchableOpacity style={styles.addButton} onPress={() => setShowAddModal(true)}>
          <View style={styles.addButtonGradient}>
            <Plus color="#ffffff" size={22} strokeWidth={2.5} />
            <Text style={styles.addButtonText}>{t('addUdhaar')}</Text>
          </View>
        </TouchableOpacity>
  
        <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.scrollContent} refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} tintColor="#10b981" />}>
          {/* Pending Customers */}
          {pendingCustomers.length > 0 && (
            <View style={styles.section}>
              <Text style={styles.sectionTitle}>{t('pendingPayments')} ({pendingCustomers.length})</Text>
              {pendingCustomers.map((customer) => (
                <View key={customer.id} style={styles.customerCard}>
                  <View style={styles.customerHeader}>
                    <View style={styles.customerAvatar}>
                      <Text style={styles.customerInitial}>{customer.customer_name.charAt(0).toUpperCase()}</Text>
                    </View>
                    <View style={styles.customerInfo}>
                      <Text style={styles.customerName}>{customer.customer_name}</Text>
                      {customer.customer_phone && (
                        <View style={styles.phoneRow}>
                          <Phone color="#64748b" size={12} />
                          <Text style={styles.customerPhone}>{customer.customer_phone}</Text>
                        </View>
                      )}
                      <Text style={styles.customerDate}>{formatDate(customer.date_created)}</Text>
                    </View>
                    <View style={styles.customerAmount}>
                      <Text style={styles.pendingAmount}>{formatCurrency(Number(customer.amount) - Number(customer.paid_amount))}</Text>
                      <Text style={styles.amountLabel}>{language === 'hi' ? 'बकाया' : 'pending'}</Text>
                    </View>
                  </View>
  
                  {customer.description && <Text style={styles.customerDesc}>{customer.description}</Text>}
  
                  <View style={styles.customerActions}>
                    {customer.customer_phone && (
                      <TouchableOpacity style={styles.whatsappAction} onPress={() => sendWhatsAppReminder(customer)}>
                        <MessageCircle color="#25D366" size={18} />
                        <Text style={styles.whatsappText}>{t('sendReminder')}</Text>
                      </TouchableOpacity>
                    )}
                    <TouchableOpacity style={styles.paymentAction} onPress={() => { setSelectedCustomer(customer); setShowPaymentModal(true); }}>
                      <IndianRupee color="#10b981" size={18} />
                      <Text style={styles.paymentText}>{language === 'hi' ? 'भुगतान' : 'Payment'}</Text>
                    </TouchableOpacity>
                  </View>
                </View>
              ))}
            </View>
          )}
  
          {/* Paid Customers */}
          {paidCustomers.length > 0 && (

              <Text style={styles.sectionTitle}>{t('paid')} ({paidCustomers.length})</Text>
              {paidCustomers.map((customer) => (
                <View key={customer.id} style={[styles.customerCard, styles.paidCard]}>
                  <View style={styles.customerHeader}>
                    <View style={[styles.customerAvatar, styles.paidAvatar]}>
                      <CheckCircle color="#10b981" size={20} />

                    <Trash2 color="#ef4444" size={16} />
                  </TouchableOpacity>
                </View>
              ))}
            </View>
          )}
  
          {customers.length === 0 && (
            <View style={styles.emptyState}>
              <Users color="#475569" size={48} strokeWidth={1.5} />
              <Text style={styles.emptyText}>{language === 'hi' ? 'कोई ग्राहक नहीं' : 'No customers yet'}</Text>
              <Text style={styles.emptySubtext}>{language === 'hi' ? 'उधार जोड़ने के लिए + पर टैप करें' : 'Tap + to add udhaar'}</Text>
            </View>
          )}
        </ScrollView>
  
        {/* Add Customer Modal */}
        <Modal visible={showAddModal} transparent animationType="slide" onRequestClose={() => setShowAddModal(false)}>
          <View style={styles.modalOverlay}>
            <View style={styles.modalContent}>
              <Text style={styles.modalTitle}>{t('addUdhaar')}</Text>
              <TextInput style={styles.modalInput} placeholder={t('customerName')} placeholderTextColor="#64748b" value={form.customer_name} onChangeText={(text) => setForm({ ...form, customer_name: text })} />
              <TextInput style={styles.modalInput} placeholder={t('customerPhone')} placeholderTextColor="#64748b" keyboardType="phone-pad" value={form.customer_phone} onChangeText={(text) => setForm({ ...form, customer_phone: text })} />
              <TextInput style={styles.modalInput} placeholder={t('amount')} placeholderTextColor="#64748b" keyboardType="decimal-pad" value={form.amount} onChangeText={(text) => setForm({ ...form, amount: text })} />

              <View style={styles.modalButtons}>
                <TouchableOpacity style={styles.modalCancelBtn} onPress={() => setShowAddModal(false)}><Text style={styles.modalCancelText}>{t('cancel')}</Text></TouchableOpacity>

          <View style={styles.modalOverlay}>
            <View style={styles.modalContent}>

        expenses: 0,
        profit: 0,

          // Re-process queued items

            await supabase.from('ledger_entries').insert(item.data);
          } else if (item.type === 'expense') {
            await supabase.from('expenses').insert(item.data);
          } else if (item.type === 'sale') {
            await supabase.from('sales').insert(item.data);
          }
        }
        await AsyncStorage.removeItem(OFFLINE_QUEUE_KEY);
      }
    } catch (err) {
      console.error('Error processing offline queue:', err);
    }
  };

  useEffect(() => {
    if (isScanning && !scanComplete) {
      Animated.loop(
        Animated.sequence([
          Animated.timing(scanLineY, { toValue: 240, duration: 1500, useNativeDriver: true }),
          Animated.timing(scanLineY, { toValue: 0, duration: 1500, useNativeDriver: true }),
        ])
      ).start();

      const timer = setTimeout(() => {
        setScanComplete(true);
        setIsScanning(false);
        setCameraActive(false);

        Animated.stagger(100,
          SAMPLE_INVOICE.items.map(() =>
            Animated.timing(itemOpacity, { toValue: 1, duration: 200, useNativeDriver: true })
          )
        ).start();

        setScannedItems(SAMPLE_INVOICE.items.map((item, index) => ({ id: `temp-${index}`, ...item })));
      }, 3000);

      return () => clearTimeout(timer);
    }
  }, [isScanning]);

  const startScanning = () => {
    setCameraActive(true);
    setIsScanning(true);
    setScanComplete(false);
    setScannedItems([]);
    itemOpacity.setValue(0);
  };

  const cancelScan = () => {
    setCameraActive(false);
    setIsScanning(false);
    setScanComplete(false);
    setScannedItems([]);
    itemOpacity.setValue(0);
  };

  const toggleCameraFacing = () => {
    setFacing(current => (current === 'back' ? 'front' : 'back'));
  };

  const formatCurrency = (amount: number) => {
    return new Intl.NumberFormat(language === 'hi' ? 'en-IN' : 'en-US', {
      style: 'currency',
      currency: 'INR',
    }).format(amount);
  };

  const grandTotal = scannedItems.reduce((sum, item) => sum + item.total, 0);

  const generateInvoiceMessage = () => {
    const itemsText = scannedItems.map(item => `${item.item_name}: ${formatCurrency(item.total)}`).join('\n');

    if (language === 'hi') {
      return `📋 *चालान विवरण*\n\nबिक्रीता: ${SAMPLE_INVOICE.vendor}\nतारीख: ${new Date().toLocaleDateString('hi-IN')}\n\n*आइटम:*\n${itemsText}\n\n━━━━━━━━━\n*कुल योग: ${formatCurrency(grandTotal)}*\n\n_The Pocket CA से भेजा गया_`;
    }

    return `📋 *Invoice Details*\n\nVendor: ${SAMPLE_INVOICE.vendor}\nDate: ${new Date().toLocaleDateString()}\n\n*Items:*\n${itemsText}\n\n━━━━━━━━━\n*Grand Total: ${formatCurrency(grandTotal)}*\n\n_Sent from The Pocket CA_`;
  };

  const shareOnWhatsApp = () => {
    if (!sharePhone) { Alert.alert(t('pleaseFillFields')); return; }
    const phone = sharePhone.replace(/\D/g, '');

    // Generate comprehensive invoice message
    let message = `*${language === 'hi' ? 'जमा रसीद' : 'RECEIPT'}*\n\n`;
    message += `${language === 'hi' ? 'दुकान: ' : 'Shop: '}The Pocket CA\n`;
    message += `${language === 'hi' ? 'तारीख: ' : 'Date: '}${new Date().toLocaleDateString('en-IN')}\n\n`;
    message += `${language === 'hi' ? 'वस्तुएं:' : 'Items:'}\n`;
    scannedItems.forEach((item, i) => {
      message += `${i + 1}. ${item.item_name} - ${formatCurrency(item.total)}\n`;
    });
    message += `\n${language === 'hi' ? 'कुल: ' : 'Total: '}${formatCurrency(grandTotal)}\n`;

    if (signature) {
      message += `\n✅ ${language === 'hi' ? 'डिजिटल हस्ताक्षरित' : 'Digitally Signed'}`;
    }

    // Add greeting card message if auto-whatsapp enabled
    if (autoWhatsappEnabled) {
      const greetings = language === 'hi'
        ? '\n\n🙏 *धन्यवाद!*\n\nहमारी दुकान में खरीदारी के लिए आपका बहुत-बहुत धन्यवाद!\n\nहम आपकी सेवा में सदैव तत्पर हैं।\n\n📍 The Pocket CA'
        : '\n\n🙏 *Thank You!*\n\nThank you so much for shopping at our store!\n\nWe are always at your service.\n\n📍 The Pocket CA';
      message += greetings;
    }

    window.open(`https://wa.me/${phone}?text=${encodeURIComponent(message)}`, '_blank');

    setShowShareModal(false);
    setSharePhone('');

    // Show success modal with PDF bill simulation
    setShowSuccessModal(true);
  };

  const closeSuccessModal = () => {
    setShowSuccessModal(false);
    router.replace('/(tabs)/dashboard');
  };

  const saveToLedger = async () => {
    if (!user) return;

    // Show signature modal first
    setShowSignatureModal(true);
  };

  const confirmSaveWithSignature = async () => {
    if (!user) return;
    setSaving(true);
    try {
      const entries = scannedItems.map(item => ({
        item_name: item.item_name, price: item.price, gst: item.gst, total: item.total,
        category: item.category, date_scanned: new Date().toISOString().split('T')[0], user_id: user.id,
        signature: signature,
      }));

      if (isOffline) {
        // Save to offline queue
        const queueData = await AsyncStorage.getItem(OFFLINE_QUEUE_KEY);
        const queue = queueData ? JSON.parse(queueData) : [];
        queue.push({ type: 'ledger', data: entries });
        queue.push({
          type: 'expense',
          data: scannedItems.map(item => ({
            description: item.item_name, amount: item.price, gst: item.gst, total: item.total,
            date: new Date().toISOString().split('T')[0], user_id: user.id, signature,
          }))
        });
        await AsyncStorage.setItem(OFFLINE_QUEUE_KEY, JSON.stringify(queue));
        Alert.alert(
          language === 'hi' ? 'ऑफ़लाइन सेव किया गया' : 'Saved Offline',
          language === 'hi' ? 'इंटरनेट से कनेक्ट होने पर डेटा सिंक हो जाएगा' : 'Data will sync when connected to internet'
        );
      } else {
        const { error } = await supabase.from('ledger_entries').insert(entries);
        if (error) throw error;

        const expenseEntries = scannedItems.map(item => ({
          description: item.item_name, amount: item.price, gst: item.gst, total: item.total,
          date: new Date().toISOString().split('T')[0], user_id: user.id, signature,
        }));

        await supabase.from('expenses').insert(expenseEntries);
      }

      setShowSignatureModal(false);
      router.replace('/(tabs)/dashboard');
    } catch (err) {
      console.error('Error saving:', err);
    } finally {
      setSaving(false);
    }
  };

  // Signature Pad Logic
  const clearSignature = () => {
    setSignature(null);
    setSignaturePaths([]);
    setCurrentSignaturePath([]);
  };

  const saveSignatureFromPad = (sig: string) => {
    setSignature(sig);
    setShowSignatureModal(false);
    // Prompt to confirm save
    confirmSaveWithSignature();
  };

  // Web canvas handlers for signature
  const handleSignatureStart = (e: any) => {
    const rect = e.target.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    setCurrentSignaturePath([{ x, y }]);
  };

  const handleSignatureMove = (e: any) => {
    if (currentSignaturePath.length === 0) return;
    const rect = e.target.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    setCurrentSignaturePath(prev => [...prev, { x, y }]);
  };

  const handleSignatureEnd = () => {
    if (currentSignaturePath.length > 1) {
      const newPaths = [...signaturePaths, currentSignaturePath];
      setSignaturePaths(newPaths);

      // Convert to data URL
      const canvas = document.createElement('canvas');
      canvas.width = CANVAS_WIDTH;
      canvas.height = CANVAS_HEIGHT;
      const ctx = canvas.getContext('2d');
      if (ctx) {
        ctx.fillStyle = '#ffffff';
        ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 2;
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';

        newPaths.forEach(path => {
          if (path.length > 1) {
            ctx.beginPath();
            ctx.moveTo(path[0].x, path[0].y);
            path.forEach((point, index) => {
              if (index > 0) ctx.lineTo(point.x, point.y);
            });
            ctx.stroke();
          }
        });
      }

      setSignature(canvas.toDataURL('image/png'));
    }
    setCurrentSignaturePath([]);
  };

  // Generate SVG path from points
  const generatePathD = (path: Point[]) => {
    if (path.length < 2) return '';
    let d = `M ${path[0].x} ${path[0].y}`;
    path.forEach((point, index) => {
      if (index > 0) d += ` L ${point.x} ${point.y}`;
    });
    return d;
  };

  // Permission screen
  if (!permission) {
    return (
      <View style={styles.container}>
        <View style={styles.loadingContainer}>
          <ActivityIndicator size="large" color="#10b981" />
        </View>
      </View>
    );
  }

  if (!permission.granted) {
    return (
      <View style={styles.container}>
        <View style={styles.header}>
          <TouchableOpacity style={styles.closeButton} onPress={() => router.back()}>
            <X color="#f8fafc" size={24} strokeWidth={2} />
          </TouchableOpacity>
          <Text style={styles.title}>{t('scanBill')}</Text>
          <View style={styles.headerRight} />
        </View>
        <View style={styles.permissionContainer}>
          <View style={styles.permissionIconContainer}>
            <Sparkles color="#10b981" size={48} strokeWidth={2} />
          </View>
          <Text style={styles.permissionTitle}>{language === 'hi' ? 'कैमरा एक्सेस चाहिए' : 'Camera Access Required'}</Text>
          <Text style={styles.permissionText}>
            {language === 'hi'
              ? 'बिल स्कैन करने के लिए, कृपया कैमरा की अनुमति दें'
              : 'To scan bills, please grant camera permission'}
          </Text>
          <TouchableOpacity style={styles.permissionButton} onPress={requestPermission}>
            <LinearGradient colors={['#10b981', '#059669']} style={styles.permissionButtonGradient}>
              <Text style={styles.permissionButtonText}>
                {language === 'hi' ? 'अनुमति दें' : 'Grant Permission'}
              </Text>
            </LinearGradient>
          </TouchableOpacity>
        </View>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      {/* Offline Banner */}
      {isOffline && (
        <View style={styles.offlineBanner}>
          <Text style={styles.offlineText}>{language === 'hi' ? 'ऑफ़लाइन मोड - कनेक्ट होने पर सिंक होगा' : 'Offline Mode - Will sync when connected'}</Text>
        </View>
      )}

      <View style={styles.header}>
        <TouchableOpacity style={styles.closeButton} onPress={() => router.back()}>
          <X color="#f8fafc" size={24} strokeWidth={2} />
        </TouchableOpacity>
        <Text style={styles.title}>{t('scanBill')}</Text>
        {cameraActive && (
          <TouchableOpacity style={styles.flipButton} onPress={toggleCameraFacing}>
            <FlipCamera2 color="#f8fafc" size={22} strokeWidth={2} />
          </TouchableOpacity>
        )}
        {!cameraActive && <View style={styles.headerRight} />}
      </View>

      <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.scrollContent}>
        <View style={styles.scannerContainer}>
          <View style={styles.scannerFrame}>
            <View style={[styles.corner, styles.topLeft]} />
            <View style={[styles.corner, styles.topRight]} />
            <View style={[styles.corner, styles.bottomLeft]} />
            <View style={[styles.corner, styles.bottomRight]} />

            {!cameraActive && !scanComplete && (
              <View style={styles.scannerPlaceholder}>
                <Sparkles color="#475569" size={48} strokeWidth={1.5} />
                <Text style={styles.scannerText}>{t('positionInvoice')}</Text>
                <TouchableOpacity style={styles.startButton} onPress={startScanning}>
                  <LinearGradient colors={['#10b981', '#059669']} style={styles.startButtonGradient}>
                    <Sparkles color="#ffffff" size={18} strokeWidth={2} />
                    <Text style={styles.startButtonText}>{t('startScanning')}</Text>
                  </LinearGradient>
                </TouchableOpacity>
              </View>
            )}

            {cameraActive && isScanning && !scanComplete && (
              <View style={styles.cameraContainer}>
                {Platform.OS === 'web' ? (
                  <View style={styles.webCameraPlaceholder}>
                    <View style={styles.invoicePreview}>
                      <Text style={styles.invoiceVendor}>Electronics Wholesale Ltd.</Text>
                      <Text style={styles.invoiceDate}>Invoice - {new Date().toLocaleDateString()}</Text>
                      <View style={styles.invoiceLineBreak} />
                      {[1, 2, 3, 4].map((_, i) => (
                        <View key={i} style={styles.invoiceLine}>
                          <View style={styles.invoiceLineBar} />
                          <View style={[styles.invoiceLineBar, styles.invoiceLineShort]} />
                        </View>
                      ))}
                    </View>
                    <Animated.View style={[styles.scanLine, { transform: [{ translateY: scanLineY }] }]} />
                    <View style={styles.scanningOverlay}>
                      <Loader color="#10b981" size={28} />
                      <Text style={styles.scanningText}>{t('scanning')}</Text>
                    </View>
                  </View>
                ) : (
                  <CameraView style={styles.camera} facing={facing}>
                    <Animated.View style={[styles.scanLine, { transform: [{ translateY: scanLineY }] }]} />
                    <View style={styles.scanningOverlay}>
                      <Loader color="#10b981" size={28} />
                      <Text style={styles.scanningText}>{t('scanning')}</Text>
                    </View>
                  </CameraView>
                )}
              </View>
            )}

            {scanComplete && (
              <View style={styles.scanSuccessContent}>
                <CheckCircle color="#10b981" size={48} strokeWidth={2} />
                <Text style={styles.scanSuccessText}>{t('scanComplete')}</Text>
                <Text style={styles.scanSuccessSubtext}>{scannedItems.length} {t('itemsDetected')}</Text>
              </View>
            )}
          </View>
        </View>

        {scanComplete && scannedItems.length > 0 && (
          <Animated.View style={[styles.ledgerContainer, { opacity: itemOpacity }]}>
            <View style={styles.ledgerHeader}>
              <FileText color="#94a3b8" size={18} strokeWidth={2} />
              <Text style={styles.ledgerTitle}>{t('extractedItems')}</Text>
            </View>

            <View style={styles.ledgerTable}>
              <View style={styles.ledgerTableHead}>
                <Text style={[styles.ledgerTableHeader, { flex: 2 }]}>{t('item')}</Text>
                <Text style={[styles.ledgerTableHeader, styles.ledgerTableHeaderRight]}>{t('price')}</Text>
                <Text style={[styles.ledgerTableHeader, styles.ledgerTableHeaderRight]}>{t('gst')}</Text>
                <Text style={[styles.ledgerTableHeader, styles.ledgerTableHeaderRight]}>{t('total')}</Text>
              </View>

              {scannedItems.map((item, index) => (
                <View key={item.id} style={[styles.ledgerTableRow, index === scannedItems.length - 1 && { borderBottomWidth: 0 }]}>
                  <Text style={[styles.ledgerTableCell, { flex: 2 }]} numberOfLines={1}>{item.item_name}</Text>
                  <Text style={[styles.ledgerTableCell, styles.ledgerTableCellRight]}>{formatCurrency(item.price)}</Text>
                  <Text style={[styles.ledgerTableCell, styles.ledgerTableCellRight]}>{formatCurrency(item.gst)}</Text>
                  <Text style={[styles.ledgerTableCell, styles.ledgerTableCellRight, styles.ledgerTableTotal]}>{formatCurrency(item.total)}</Text>
                </View>
              ))}

              <View style={styles.grandTotalRow}>
                <Text style={[styles.grandTotalLabel, { flex: 2 }]}>{t('grandTotal')}</Text>
                <Text style={styles.grandTotalValue}>{formatCurrency(grandTotal)}</Text>
              </View>
            </View>

            {/* Signature Preview */}
            {signature && (
              <View style={styles.signaturePreview}>
                <PenLine color="#10b981" size={16} />
                <Text style={styles.signatureLabel}>{language === 'hi' ? 'हस्ताक्षरित' : 'Signed'}</Text>
              </View>
            )}

            <TouchableOpacity style={styles.whatsappShareBtn} onPress={() => setShowShareModal(true)}>
              <MessageCircle color="#25D366" size={20} strokeWidth={2} />
              <Text style={styles.whatsappShareText}>{t('shareOnWhatsApp')}</Text>
            </TouchableOpacity>

            <View style={styles.actionButtons}>
              <TouchableOpacity style={styles.cancelBtn} onPress={cancelScan}>
                <Text style={styles.cancelBtnText}>{t('scanAgain')}</Text>
              </TouchableOpacity>

              <TouchableOpacity style={styles.saveBtn} onPress={saveToLedger} disabled={saving}>
                <LinearGradient colors={['#10b981', '#059669']} style={styles.saveBtnGradient}>
                  {saving ? (
                    <Text style={styles.saveBtnText}>{t('saving')}</Text>
                  ) : (
                    <>
                      <PenLine color="#ffffff" size={18} strokeWidth={2.5} />
                      <Text style={styles.saveBtnText}>{language === 'hi' ? 'हस्ताक्षर और सेव' : 'Sign & Save'}</Text>
                    </>
                  )}
                </LinearGradient>
              </TouchableOpacity>
            </View>
          </Animated.View>
        )}
      </ScrollView>

      {/* WhatsApp Share Modal */}
      <Modal visible={showShareModal} transparent animationType="slide" onRequestClose={() => setShowShareModal(false)}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>{t('shareOnWhatsApp')}</Text>
            <TextInput style={styles.modalInput} placeholder={t('customerPhone')} placeholderTextColor="#64748b" keyboardType="phone-pad" value={sharePhone} onChangeText={setSharePhone} />
            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.modalCancelBtn} onPress={() => setShowShareModal(false)}>
                <Text style={styles.modalCancelText}>{t('cancel')}</Text>
              </TouchableOpacity>
              <TouchableOpacity style={styles.modalSaveBtn} onPress={shareOnWhatsApp}>
                <MessageCircle color="#ffffff" size={18} />
                <Text style={styles.modalSaveText}>{t('shareOnWhatsApp')}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>

      {/* Signature Modal */}
      <Modal visible={showSignatureModal} transparent animationType="slide" onRequestClose={() => setShowSignatureModal(false)}>
        <View style={styles.modalOverlay}>
          <View style={styles.signatureModalContent}>
            <Text style={styles.signatureTitle}>{language === 'hi' ? 'डिजिटल हस्ताक्षर' : 'Digital Signature'}</Text>
            <Text style={styles.signatureHint}>{language === 'hi' ? 'नीचे अपना हस्ताक्षर करें' : 'Sign below to confirm'}</Text>

            {/* Web Canvas Signature Pad */}
            {typeof document !== 'undefined' && (
              <View style={styles.signatureWrapper}>
                <svg
                  width={CANVAS_WIDTH}
                  height={CANVAS_HEIGHT}
                  style={styles.signatureCanvas}
                  onMouseDown={handleSignatureStart}
                  onMouseMove={handleSignatureMove}
                  onMouseUp={handleSignatureEnd}
                  onMouseLeave={handleSignatureEnd}
                >
                  <rect width={CANVAS_WIDTH} height={CANVAS_HEIGHT} fill="#ffffff" />
                  {signaturePaths.map((path, index) => (
                    <path key={index} d={generatePathD(path)} fill="none" stroke="#ffffff" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
                  ))}
                  {currentSignaturePath.length > 1 && (
                    <path d={generatePathD(currentSignaturePath)} fill="none" stroke="#ffffff" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
                  )}
                </svg>
              </View>
            )}

            <View style={styles.signatureActions}>
              <TouchableOpacity style={styles.clearSigBtn} onPress={clearSignature}>
                <Text style={styles.clearSigText}>{language === 'hi' ? 'साफ़ करें' : 'Clear'}</Text>
              </TouchableOpacity>
              <TouchableOpacity style={styles.confirmSigBtn} onPress={confirmSaveWithSignature}>
                <CheckCircle color="#ffffff" size={18} />
                <Text style={styles.confirmSigText}>{language === 'hi' ? 'कन्फर्म करें' : 'Confirm'}</Text>
              </TouchableOpacity>
            </View>

            <TouchableOpacity style={styles.skipSigBtn} onPress={() => { setShowSignatureModal(false); confirmSaveWithSignature(); }}>
              <Text style={styles.skipSigText}>{language === 'hi' ? 'हस्ताक्षर छोड़ें' : 'Skip Signature'}</Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>

      {/* Success Modal - WhatsApp Billing & Greeting Simulation */}
      <Modal visible={showSuccessModal} transparent animationType="fade" onRequestClose={closeSuccessModal}>
        <View style={styles.successModalOverlay}>
          <View style={styles.successModalContent}>
            <View style={styles.successIconContainer}>
              <PartyPopper color="#ffffff" size={40} />
            </View>
            <Text style={styles.successTitle}>{language === 'hi' ? 'बिल भेजा गया!' : 'Bill Sent Successfully!'}</Text>

            <View style={styles.successFeatures}>
              <View style={styles.successFeature}>
                <View style={[styles.successFeatureIcon, { backgroundColor: 'rgba(37, 211, 102, 0.1)' }]}>
                  <MessageCircle color="#25D366" size={20} />
                </View>
                <Text style={styles.successFeatureText}>{language === 'hi' ? 'PDF बिल WhatsApp पर भेजा गया' : 'PDF Bill sent on WhatsApp'}</Text>
              </View>

              {autoWhatsappEnabled && (
                <View style={styles.successFeature}>
                  <View style={[styles.successFeatureIcon, { backgroundColor: 'rgba(236, 72, 153, 0.1)' }]}>
                    <Gift color="#ec4899" size={20} />
                  </View>
                  <Text style={styles.successFeatureText}>{language === 'hi' ? 'शुभकामना कार्ड भी भेजा गया' : 'Greeting card also sent'}</Text>
                </View>
              )}

              {loyaltyEnabled && (
                <View style={styles.successFeature}>
                  <View style={[styles.successFeatureIcon, { backgroundColor: 'rgba(245, 158, 11, 0.1)' }]}>
                    <Gift color="#f59e0b" size={20} />
                  </View>
                  <Text style={styles.successFeatureText}>{language === 'hi' ? `लॉयल्टी पॉइंट अर्जित: ${Math.floor(grandTotal / 100)}` : `Loyalty points earned: ${Math.floor(grandTotal / 100)}`}</Text>
                </View>
              )}
            </View>

            <TouchableOpacity style={styles.successCloseBtn} onPress={closeSuccessModal}>
              <CheckCircle color="#ffffff" size={20} />
              <Text style={styles.successCloseText}>{language === 'hi' ? 'ठीक है' : 'Done'}</Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>
    </View>
  );
}

  container: { flex: 1, backgroundColor: '#f8fafc' },

  offlineBanner: { backgroundColor: 'rgba(245, 158, 11, 0.1)', paddingVertical: 8, paddingHorizontal: 16, marginHorizontal: 20, borderRadius: 8, marginBottom: 8 },
  offlineText: { fontSize: 12, color: '#b45309', textAlign: 'center' },
  header: { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between', paddingHorizontal: 20, paddingTop: 60, paddingBottom: 16 },
  closeButton: { width: 44, height: 44, borderRadius: 14, backgroundColor: '#ffffff', justifyContent: 'center', alignItems: 'center', borderWidth: 1, borderColor: 'rgba(0, 0, 0, 0.1)' },
  flipButton: { width: 44, height: 44, borderRadius: 14, backgroundColor: '#ffffff', justifyContent: 'center', alignItems: 'center', borderWidth: 1, borderColor: 'rgba(0, 0, 0, 0.1)' },
  title: { fontSize: 22, fontWeight: '700', color: '#1e293b' },

import { useState, useEffect } from 'react';
import { View, Text, StyleSheet, TouchableOpacity, Alert, Switch, ScrollView } from 'react-native';
import { useRouter } from 'expo-router';
import { User, Shield, LogOut, ChevronRight, Calculator, HelpCircle, Globe, Bell, Clock, MessageCircle, Gift, Settings } from 'lucide-react-native';
import { useAuth } from '@/contexts/AuthContext';
import { useI18n } from '@/contexts/I18nContext';
import { supabase } from '@/lib/supabase';

interface NotificationSettings {
  daily_summary_enabled: boolean;
  daily_summary_time: string;
  low_stock_alerts: boolean;
}

interface UserSettings {
  auto_whatsapp_billing: boolean;
  loyalty_points_enabled: boolean;
  points_per_rupee: number;
  rupee_per_point: number;
}

export default function SettingsScreen() {
  const { user, signOut } = useAuth();
  const { t, language, toggleLanguage } = useI18n();
  const router = useRouter();
  const [notificationSettings, setNotificationSettings] = useState<NotificationSettings>({
    daily_summary_enabled: true,
    daily_summary_time: '20:00',
    low_stock_alerts: true,
  });
  const [userSettings, setUserSettings] = useState<UserSettings>({
    auto_whatsapp_billing: false,
    loyalty_points_enabled: true,
    points_per_rupee: 100,
    rupee_per_point: 1,
  });
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchSettings();
  }, [user]);

  const fetchSettings = async () => {
    if (!user) return;
    try {
      const [notifRes, userRes] = await Promise.all([
        supabase.from('notification_settings').select('*').eq('user_id', user.id).maybeSingle(),
        supabase.from('user_settings').select('*').eq('user_id', user.id).maybeSingle(),
      ]);

      if (notifRes.data) {
        setNotificationSettings({
          daily_summary_enabled: notifRes.data.daily_summary_enabled,
          daily_summary_time: notifRes.data.daily_summary_time,
          low_stock_alerts: notifRes.data.low_stock_alerts,
        });
      } else {
        await supabase.from('notification_settings').insert({
          user_id: user.id,
          daily_summary_enabled: true,
          daily_summary_time: '20:00',
          low_stock_alerts: true,
        });
      }

      if (userRes.data) {
        setUserSettings({
          auto_whatsapp_billing: userRes.data.auto_whatsapp_billing ?? false,
          loyalty_points_enabled: userRes.data.loyalty_points_enabled ?? true,
          points_per_rupee: userRes.data.points_per_rupee || 100,
          rupee_per_point: userRes.data.rupee_per_point || 1,
        });
      } else {
        await supabase.from('user_settings').insert({
          user_id: user.id,
          auto_whatsapp_billing: false,
          loyalty_points_enabled: true,
          points_per_rupee: 100,
          rupee_per_point: 1,
        });
      }
    } catch (err) {
      console.error('Error fetching settings:', err);
    } finally {
      setLoading(false);
    }
  };

  const updateNotificationSettings = async (key: keyof NotificationSettings, value: boolean | string) => {
    if (!user) return;
    const newSettings = { ...notificationSettings, [key]: value };
    setNotificationSettings(newSettings);
    try {
      await supabase
        .from('notification_settings')
        .update({ [key]: value })
        .eq('user_id', user.id);
    } catch (err) {
      console.error('Error updating settings:', err);
    }
  };

  const updateUserSettings = async (key: keyof UserSettings, value: boolean | number) => {
    if (!user) return;
    const newSettings = { ...userSettings, [key]: value };
    setUserSettings(newSettings);
    try {
      await supabase
        .from('user_settings')
        .update({ [key]: value })
        .eq('user_id', user.id);
    } catch (err) {
      console.error('Error updating user settings:', err);
    }
  };

  const handleSignOut = async () => {
    await signOut();
    router.replace('/login');
  };

  const confirmSignOut = () => {
    Alert.alert(
      t('signOut'),
      t('confirmSignOut'),
      [
        { text: t('no'), style: 'cancel' },
        { text: t('yes'), style: 'destructive', onPress: handleSignOut }
      ]
    );
  };

  return (
    <View style={styles.container}>
      <View style={styles.header}>
        <Text style={styles.title}>{t('settings')}</Text>
        <TouchableOpacity style={styles.languageBtn} onPress={toggleLanguage}>
          <Globe color="#94a3b8" size={18} />
          <Text style={styles.languageText}>{language === 'en' ? 'हिंदी' : 'English'}</Text>
        </TouchableOpacity>
      </View>

      <ScrollView
        style={styles.scrollView}
        contentContainerStyle={styles.scrollContent}
        showsVerticalScrollIndicator={false}
      >
        {/* Profile Section */}
        <View style={styles.profileCard}>
          <View style={styles.profileIcon}>
            <Calculator color="#10b981" size={32} strokeWidth={2} />
          </View>
          <View style={styles.profileInfo}>
            <Text style={styles.profileName}>The Pocket CA</Text>
            <Text style={styles.profileEmail}>{user?.email || 'No email'}</Text>
          </View>
        </View>

        {/* Language Toggle */}
        <TouchableOpacity style={styles.settingItem} onPress={toggleLanguage}>
          <View style={styles.settingIconBg}>
            <Globe color="#64748b" size={20} strokeWidth={2} />
          </View>
          <View style={styles.settingText}>
            <Text style={styles.settingTitle}>{t('language')}</Text>
            <Text style={styles.settingSubtitle}>{language === 'en' ? t('hindi') : t('english')}</Text>
          </View>
          <ChevronRight color="#475569" size={20} strokeWidth={2} />
        </TouchableOpacity>

        {/* Notifications Section */}
        <View style={styles.sectionHeader}>
          <Bell color="#94a3b8" size={18} />
          <Text style={styles.sectionTitle}>{t('notifications')}</Text>
        </View>

        <View style={styles.settingsList}>
          <View style={[styles.settingItem, styles.settingItemFirst]}>
            <View style={styles.settingIconBg}>
              <Bell color="#64748b" size={20} strokeWidth={2} />
            </View>
            <View style={styles.settingText}>
              <Text style={styles.settingTitle}>{t('enableDailySummary')}</Text>
              <Text style={styles.settingSubtitle}>{t('dailySummary')}</Text>
            </View>
            <Switch
              value={notificationSettings.daily_summary_enabled}
              onValueChange={(value) => updateNotificationSettings('daily_summary_enabled', value)}
              trackColor={{ false: '#475569', true: '#10b981' }}
              thumbColor="#f8fafc"
            />
          </View>

          <View style={styles.settingItem}>
            <View style={styles.settingIconBg}>
              <Clock color="#64748b" size={20} strokeWidth={2} />
            </View>
            <View style={styles.settingText}>
              <Text style={styles.settingTitle}>{t('dailySummaryTime')}</Text>
              <Text style={styles.settingSubtitle}>{notificationSettings.daily_summary_time}</Text>
            </View>
            <ChevronRight color="#475569" size={20} strokeWidth={2} />
          </View>

          <View style={[styles.settingItem, styles.settingItemLast]}>
            <View style={styles.settingIconBg}>
              <Shield color="#64748b" size={20} strokeWidth={2} />
            </View>
            <View style={styles.settingText}>
              <Text style={styles.settingTitle}>{t('enableLowStockAlerts')}</Text>
              <Text style={styles.settingSubtitle}>{t('lowStockAlert')}</Text>
            </View>
            <Switch
              value={notificationSettings.low_stock_alerts}
              onValueChange={(value) => updateNotificationSettings('low_stock_alerts', value)}
              trackColor={{ false: '#475569', true: '#10b981' }}
              thumbColor="#f8fafc"
            />
          </View>
        </View>

        {/* WhatsApp Billing Section */}
        <View style={styles.sectionHeader}>
          <MessageCircle color="#25D366" size={18} />
          <Text style={styles.sectionTitle}>{language === 'hi' ? 'व्हाट्सएप बिलिंग' : 'WhatsApp Billing'}</Text>
        </View>

        <View style={styles.settingsList}>
          <View style={[styles.settingItem, styles.settingItemFirst, styles.settingItemLast]}>
            <View style={[styles.settingIconBg, { backgroundColor: 'rgba(37, 211, 102, 0.1)' }]}>
              <MessageCircle color="#25D366" size={20} strokeWidth={2} />
            </View>
            <View style={styles.settingText}>
              <Text style={styles.settingTitle}>{language === 'hi' ? 'ऑटो व्हाट्सएप बिल' : 'Auto WhatsApp Receipts'}</Text>
              <Text style={styles.settingSubtitle}>{language === 'hi' ? 'बिल भेजने पर ग्राहक को व्हाट्सएप पर PDF भेजें' : 'Send PDF bill to customer on WhatsApp'}</Text>
            </View>
            <Switch
              value={userSettings.auto_whatsapp_billing}
              onValueChange={(value) => updateUserSettings('auto_whatsapp_billing', value)}
              trackColor={{ false: '#475569', true: '#25D366' }}
              thumbColor="#f8fafc"
            />
          </View>
        </View>

        {/* Loyalty Points Section */}
        <View style={styles.sectionHeader}>
          <Gift color="#ec4899" size={18} />
          <Text style={styles.sectionTitle}>{language === 'hi' ? 'लॉयल्टी प्रणाली' : 'Loyalty Program'}</Text>
        </View>

        <View style={styles.settingsList}>
          <View style={[styles.settingItem, styles.settingItemFirst, styles.settingItemLast]}>
            <View style={[styles.settingIconBg, { backgroundColor: 'rgba(236, 72, 153, 0.1)' }]}>
              <Gift color="#ec4899" size={20} strokeWidth={2} />
            </View>
            <View style={styles.settingText}>
              <Text style={styles.settingTitle}>{language === 'hi' ? 'लॉयल्टी पॉइंट्स' : 'Loyalty Points'}</Text>
              <Text style={styles.settingSubtitle}>{language === 'hi' ? `₹${userSettings.points_per_rupee} खरीद पर 1 पॉइंट` : `1 point per ₹${userSettings.points_per_rupee} spent`}</Text>
            </View>
            <Switch
              value={userSettings.loyalty_points_enabled}
              onValueChange={(value) => updateUserSettings('loyalty_points_enabled', value)}
              trackColor={{ false: '#475569', true: '#ec4899' }}
              thumbColor="#f8fafc"
            />
          </View>
        </View>

        {/* Other Settings */}
        <View style={styles.sectionHeader}>
          <User color="#94a3b8" size={18} />
          <Text style={styles.sectionTitle}>{t('account')}</Text>
        </View>

        <View style={styles.settingsList}>
          <TouchableOpacity style={[styles.settingItem, styles.settingItemFirst]} activeOpacity={0.7}>
            <View style={styles.settingIconBg}>
              <User color="#64748b" size={20} strokeWidth={2} />

            <View style={styles.settingText}>
              <Text style={styles.settingTitle}>{t('account')}</Text>

        .from('purchase_orders')

      ]);

            color: cat.color,

              placeholder={language === 'hi' ? 'श्रेणी का नाम *' : 'Category name *'}
              placeholderTextColor="#94a3b8"
              value={categoryForm.name}
              onChangeText={(text) => setCategoryForm({ ...categoryForm, name: text })}
            />

            <Text style={styles.inputLabel}>{language === 'hi' ? 'रंग चुनें' : 'Select Color'}</Text>
            <View style={styles.colorPicker}>
              {['#6366f1', '#f59e0b', '#ec4899', '#3b82f6', '#8b5cf6', '#10b981', '#ef4444', '#64748b'].map((color) => (
                <TouchableOpacity
                  key={color}
                  style={[styles.colorOption, { backgroundColor: color }, categoryForm.color === color && styles.colorOptionSelected]}
                  onPress={() => setCategoryForm({ ...categoryForm, color })}
                />
              ))}
            </View>

            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.cancelBtn} onPress={() => setShowCategoryModal(false)}>
                <Text style={styles.cancelBtnText}>{language === 'hi' ? 'रद्द करें' : 'Cancel'}</Text>
              </TouchableOpacity>
              <TouchableOpacity style={styles.submitBtn} onPress={handleAddCategory}>
                <Text style={styles.submitBtnText}>{language === 'hi' ? 'जोड़ें' : 'Add'}</Text>

  );
}

expiry-tracker.tsx
import { useState, useEffect } from 'react';
import { View, Text, StyleSheet, ScrollView, TouchableOpacity, Modal, TextInput, Alert, RefreshControl } from 'react-native';
import { useRouter, useLocalSearchParams } from 'expo-router';
import { LinearGradient } from 'expo-linear-gradient';
import { ArrowLeft, Plus, Clock, AlertTriangle, Trash2, Calendar, Package, Search } from 'lucide-react-native';
import { useAuth } from '@/contexts/AuthContext';
import { useI18n } from '@/contexts/I18nContext';
import { supabase } from '@/lib/supabase';

interface InventoryItem {
  id: string;
  name: string;
  quantity: number;
  unit: string;
  expiry_date: string | null;
  purchase_price: number | null;
  category: string | null;
  is_dead_stock: boolean;
}

export default function ExpiryTrackerScreen() {
  const { user } = useAuth();
  const { t, language } = useI18n();
  const router = useRouter();
  const [items, setItems] = useState<InventoryItem[]>([]);
  const [filteredItems, setFilteredItems] = useState<InventoryItem[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [activeTab, setActiveTab] = useState<'all' | 'expiring' | 'expired' | 'dead'>('all');
  const [showAddModal, setShowAddModal] = useState(false);
  const [form, setForm] = useState({ name: '', quantity: '', unit: 'pcs', expiry_date: '', purchase_price: '', category: '' });

  useEffect(() => {
    fetchItems();
  }, [user]);

  useEffect(() => {
    filterItems();
  }, [items, activeTab, searchQuery]);

  const fetchItems = async () => {
    if (!user) return;
    setLoading(true);
    try {
      const { data, error } = await supabase
        .from('inventory_items')
        .select('*')
        .eq('user_id', user.id)
        .order('expiry_date', { ascending: true });

      if (error) throw error;
      setItems(data || []);
    } catch (err) {
      console.error('Error fetching items:', err);
    } finally {
      setLoading(false);
    }
  };

  const filterItems = () => {
    const today = new Date();
    const fifteenDaysFromNow = new Date(today.getTime() + 15 * 24 * 60 * 60 * 1000);

    let filtered = items;

    if (searchQuery) {
      filtered = filtered.filter(item =>
        item.name.toLowerCase().includes(searchQuery.toLowerCase())
      );
    }

    switch (activeTab) {
      case 'expiring':
        filtered = filtered.filter(item => {
          if (!item.expiry_date || item.is_dead_stock) return false;
          const expiry = new Date(item.expiry_date);
          return expiry >= today && expiry <= fifteenDaysFromNow;
        });
        break;
      case 'expired':
        filtered = filtered.filter(item => {
          if (!item.expiry_date || item.is_dead_stock) return false;
          const expiry = new Date(item.expiry_date);
          return expiry < today;
        });
        break;
      case 'dead':
        filtered = filtered.filter(item => item.is_dead_stock);
        break;
    }

    setFilteredItems(filtered);
  };

  const getDaysUntilExpiry = (expiryDate: string): number => {
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const expiry = new Date(expiryDate);
    expiry.setHours(0, 0, 0, 0);
    return Math.ceil((expiry.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
  };

  const getExpiryStatus = (item: InventoryItem): { color: string; text: string } => {
    if (item.is_dead_stock) return { color: '#64748b', text: language === 'hi' ? 'मृत स्टॉक' : 'Dead Stock' };
    if (!item.expiry_date) return { color: '#64748b', text: language === 'hi' ? 'कोई तारीख नहीं' : 'No date' };

    const days = getDaysUntilExpiry(item.expiry_date);
    if (days < 0) return { color: '#ef4444', text: language === 'hi' ? 'एक्सपायर हो चुका' : 'Expired' };
    if (days <= 15) return { color: '#f59e0b', text: `${days} ${language === 'hi' ? 'दिन बाकी' : 'days left'}` };
    return { color: '#10b981', text: `${days} ${language === 'hi' ? 'दिन बाकी' : 'days left'}` };
  };

  const handleAddItem = async () => {
    if (!user || !form.name || !form.quantity) return;

    try {
      const { error } = await supabase.from('inventory_items').insert({
        user_id: user.id,
        name: form.name,
        quantity: parseInt(form.quantity),
        unit: form.unit,
        expiry_date: form.expiry_date || null,
        purchase_price: form.purchase_price ? parseFloat(form.purchase_price) : null,
        category: form.category || null,
      });

      if (error) throw error;
      setShowAddModal(false);
      setForm({ name: '', quantity: '', unit: 'pcs', expiry_date: '', purchase_price: '', category: '' });
      fetchItems();
    } catch (err) {
      console.error('Error adding item:', err);
      Alert.alert('Error', 'Failed to add item');
    }
  };

  const handleMarkAsDead = async (item: InventoryItem) => {
    if (!user) return;

    try {
      const { error } = await supabase
        .from('inventory_items')
        .update({ is_dead_stock: true })
        .eq('id', item.id);

      if (error) throw error;
      fetchItems();
    } catch (err) {
      console.error('Error updating item:', err);
    }
  };

  const handleDelete = async (id: string) => {
    Alert.alert(
      language === 'hi' ? 'हटाएं' : 'Delete',
      language === 'hi' ? 'क्या आप इस वस्तु को हटाना चाहते हैं?' : 'Are you sure you want to delete this item?',
      [
        { text: language === 'hi' ? 'नहीं' : 'No', style: 'cancel' },
        {
          text: language === 'hi' ? 'हां' : 'Yes',
          style: 'destructive',
          onPress: async () => {
            try {
              await supabase.from('inventory_items').delete().eq('id', id);
              fetchItems();
            } catch (err) {
              console.error('Error deleting:', err);
            }
          },
        },
      ]
    );
  };

  const onRefresh = async () => {
    setRefreshing(true);
    await fetchItems();
    setRefreshing(false);
  };

  const stats = {
    total: items.length,
    expiring: items.filter(i => {
      if (!i.expiry_date || i.is_dead_stock) return false;
      const days = getDaysUntilExpiry(i.expiry_date);
      return days >= 0 && days <= 15;
    }).length,
    expired: items.filter(i => {
      if (!i.expiry_date || i.is_dead_stock) return false;
      return getDaysUntilExpiry(i.expiry_date) < 0;
    }).length,
    dead: items.filter(i => i.is_dead_stock).length,
  };

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.header}>
        <TouchableOpacity style={styles.backBtn} onPress={() => router.back()}>
          <ArrowLeft color="#1e293b" size={24} strokeWidth={2} />
        </TouchableOpacity>
        <Text style={styles.headerTitle}>{language === 'hi' ? 'एक्सपायरी ट्रैकर' : 'Expiry Tracker'}</Text>
        <TouchableOpacity style={styles.addBtn} onPress={() => setShowAddModal(true)}>
          <Plus color="#ffffff" size={22} strokeWidth={2.5} />
        </TouchableOpacity>
      </View>

      {/* Stats Cards */}
      <View style={styles.statsRow}>
        <TouchableOpacity
          style={[styles.statCard, activeTab === 'expiring' && styles.statCardActive]}
          onPress={() => setActiveTab('expiring')}
        >
          <Clock color={activeTab === 'expiring' ? '#f59e0b' : '#94a3b8'} size={18} />
          <Text style={[styles.statValue, activeTab === 'expiring' && { color: '#f59e0b' }]}>{stats.expiring}</Text>
          <Text style={styles.statLabel}>{language === 'hi' ? 'जल्द एक्सपायर' : 'Expiring'}</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.statCard, activeTab === 'expired' && styles.statCardDanger]}
          onPress={() => setActiveTab('expired')}
        >
          <AlertTriangle color={activeTab === 'expired' ? '#ef4444' : '#94a3b8'} size={18} />
          <Text style={[styles.statValue, activeTab === 'expired' && { color: '#ef4444' }]}>{stats.expired}</Text>
          <Text style={styles.statLabel}>{language === 'hi' ? 'एक्सपायर' : 'Expired'}</Text>
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.statCard, activeTab === 'dead' && styles.statCardMuted]}
          onPress={() => setActiveTab('dead')}
        >
          <Package color={activeTab === 'dead' ? '#64748b' : '#94a3b8'} size={18} />
          <Text style={[styles.statValue, activeTab === 'dead' && { color: '#64748b' }]}>{stats.dead}</Text>
          <Text style={styles.statLabel}>{language === 'hi' ? 'मृत स्टॉक' : 'Dead'}</Text>
        </TouchableOpacity>
      </View>

      {/* Search */}
      <View style={styles.searchContainer}>
        <Search color="#94a3b8" size={18} />
        <TextInput
          style={styles.searchInput}
          placeholder={language === 'hi' ? 'वस्तु खोजें...' : 'Search items...'}
          placeholderTextColor="#94a3b8"
          value={searchQuery}
          onChangeText={setSearchQuery}
        />
      </View>

      {/* Tab Buttons */}
      <View style={styles.tabsContainer}>
        {['all', 'expiring', 'expired', 'dead'].map((tab) => (
          <TouchableOpacity
            key={tab}
            style={[styles.tab, activeTab === tab && styles.tabActive]}
            onPress={() => setActiveTab(tab as any)}
          >
            <Text style={[styles.tabText, activeTab === tab && styles.tabTextActive]}>
              {tab === 'all' ? (language === 'hi' ? 'सभी' : 'All') :
               tab === 'expiring' ? (language === 'hi' ? 'जल्द एक्सपायर' : 'Expiring') :
               tab === 'expired' ? (language === 'hi' ? 'एक्सपायर' : 'Expired') :
               (language === 'hi' ? 'मृत स्टॉक' : 'Dead Stock')}
            </Text>
          </TouchableOpacity>
        ))}
      </View>

      {/* Red Alert Banner */}
      {stats.expired > 0 && activeTab !== 'expired' && (
        <View style={styles.alertBanner}>
          <AlertTriangle color="#ef4444" size={20} />
          <Text style={styles.alertText}>
            {language === 'hi' ? `चेतावनी! ${stats.expired} वस्तुएं एक्सपायर हो चुकी हैं!` : `Alert! ${stats.expired} items have expired!`}
          </Text>
        </View>
      )}

      {/* Items List */}
      <ScrollView
        style={styles.list}
        showsVerticalScrollIndicator={false}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} tintColor="#10b981" />}
      >
        {filteredItems.length === 0 ? (
          <View style={styles.emptyState}>
            <Package color="#cbd5e1" size={48} />
            <Text style={styles.emptyText}>
              {language === 'hi' ? 'कोई वस्तु नहीं मिली' : 'No items found'}
            </Text>
          </View>
        ) : (
          filteredItems.map((item) => {
            const status = getExpiryStatus(item);
            return (
              <View key={item.id} style={styles.itemCard}>
                <View style={styles.itemMain}>
                  <View style={[styles.itemIcon, { backgroundColor: `${status.color}15` }]}>
                    <Package color={status.color} size={20} />
                  </View>
                  <View style={styles.itemInfo}>
                    <Text style={styles.itemName}>{item.name}</Text>
                    <Text style={styles.itemMeta}>
                      {item.quantity} {item.unit}
                      {item.category && ` • ${item.category}`}
                    </Text>
                  </View>
                  <View style={[styles.statusBadge, { backgroundColor: `${status.color}15` }]}>
                    <Text style={[styles.statusText, { color: status.color }]}>{status.text}</Text>
                  </View>
                </View>
                {item.expiry_date && (
                  <View style={styles.itemFooter}>
                    <Calendar color="#94a3b8" size={14} />
                    <Text style={styles.dateText}>
                      {language === 'hi' ? 'एक्सपायरी: ' : 'Expires: '}
                      {new Date(item.expiry_date).toLocaleDateString('en-IN')}
                    </Text>
                    {!item.is_dead_stock && (
                      <TouchableOpacity
                        style={styles.deadStockBtn}
                        onPress={() => handleMarkAsDead(item)}
                      >
                        <Text style={styles.deadStockText}>
                          {language === 'hi' ? 'मृत स्टॉक चिह्नित करें' : 'Mark Dead'}
                        </Text>
                      </TouchableOpacity>
                    )}
                    <TouchableOpacity onPress={() => handleDelete(item.id)} style={styles.deleteBtn}>
                      <Trash2 color="#ef4444" size={16} />
                    </TouchableOpacity>
                  </View>
                )}
              </View>
            );
          })
        )}
        <View style={styles.bottomSpace} />
      </ScrollView>

      {/* Add Modal */}
      <Modal visible={showAddModal} transparent animationType="slide" onRequestClose={() => setShowAddModal(false)}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>{language === 'hi' ? 'नई वस्तु जोड़ें' : 'Add New Item'}</Text>

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'वस्तु का नाम *' : 'Item name *'}
              placeholderTextColor="#94a3b8"
              value={form.name}
              onChangeText={(text) => setForm({ ...form, name: text })}
            />

            <View style={styles.inputRow}>
              <TextInput
                style={[styles.input, { flex: 1 }]}
                placeholder={language === 'hi' ? 'मात्रा *' : 'Quantity *'}
                placeholderTextColor="#94a3b8"
                keyboardType="numeric"
                value={form.quantity}
                onChangeText={(text) => setForm({ ...form, quantity: text })}
              />
              <TextInput
                style={[styles.input, { flex: 1 }]}
                placeholder={language === 'hi' ? 'इकाई' : 'Unit'}

import { useState, useEffect } from 'react';
import { View, Text, StyleSheet, ScrollView, TouchableOpacity, Modal, TextInput, Alert, RefreshControl } from 'react-native';
import { useRouter } from 'expo-router';
import { ArrowLeft, Plus, Users, Phone, Calendar, IndianRupee, UserPlus, Check, X, AlertCircle, ChevronDown, ChevronUp, Trash2, Clock } from 'lucide-react-native';
import { useAuth } from '@/contexts/AuthContext';
import { useI18n } from '@/contexts/I18nContext';
import { supabase } from '@/lib/supabase';

interface Employee {
  id: string;
  name: string;
  phone: string | null;
  role: string;
  monthly_salary: number;
  joining_date: string | null;
  is_active: boolean;
}

interface AttendanceRecord {
  id: string;
  employee_id: string;
  date: string;
  status: 'present' | 'absent' | 'half_day' | 'leave';
  check_in_time: string | null;
  check_out_time: string | null;
}

interface SalaryAdvance {
  id: string;
  employee_id: string;
  amount: number;
  date_given: string;
  reason: string | null;
  is_deducted: boolean;
  deduction_month: string | null;
}

export default function StaffScreen() {
  const { user } = useAuth();
  const { language } = useI18n();
  const router = useRouter();
  const [employees, setEmployees] = useState<Employee[]>([]);
  const [attendance, setAttendance] = useState<AttendanceRecord[]>([]);
  const [advances, setAdvances] = useState<SalaryAdvance[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [activeTab, setActiveTab] = useState<'staff' | 'attendance' | 'salary'>('staff');
  const [selectedEmployee, setSelectedEmployee] = useState<Employee | null>(null);
  const [showAddModal, setShowAddModal] = useState(false);
  const [showAdvanceModal, setShowAdvanceModal] = useState(false);
  const [expandedEmployee, setExpandedEmployee] = useState<string | null>(null);
  const [form, setForm] = useState({ name: '', phone: '', role: 'Staff', monthly_salary: '' });
  const [advanceForm, setAdvanceForm] = useState({ amount: '', reason: '' });

  const today = new Date().toISOString().split('T')[0];
  const currentMonth = new Date().toISOString().slice(0, 7);

  useEffect(() => {
    fetchData();
  }, [user]);

  const fetchData = async () => {
    if (!user) return;
    setLoading(true);
    try {
      const [employeesRes, attendanceRes, advancesRes] = await Promise.all([
        supabase.from('employees').select('*').eq('user_id', user.id).eq('is_active', true).order('name'),
        supabase.from('attendance').select('*').eq('user_id', user.id).eq('date', today),
        supabase.from('salary_advances').select('*').eq('user_id', user.id),
      ]);

      setEmployees(employeesRes.data || []);
      setAttendance(attendanceRes.data || []);
      setAdvances(advancesRes.data || []);
    } catch (err) {
      console.error('Error fetching data:', err);
    } finally {
      setLoading(false);
    }
  };

  const onRefresh = async () => {
    setRefreshing(true);
    await fetchData();
    setRefreshing(false);
  };

  const handleAddEmployee = async () => {
    if (!user || !form.name || !form.monthly_salary) return;

    try {
      const { error } = await supabase.from('employees').insert({
        user_id: user.id,
        name: form.name,
        phone: form.phone || null,
        role: form.role || 'Staff',
        monthly_salary: parseFloat(form.monthly_salary),
        joining_date: today,
        is_active: true,
      });

      if (error) throw error;
      setShowAddModal(false);
      setForm({ name: '', phone: '', role: 'Staff', monthly_salary: '' });
      fetchData();
    } catch (err) {
      console.error('Error adding employee:', err);
      Alert.alert('Error', 'Failed to add employee');
    }
  };

  const handleMarkAttendance = async (employeeId: string, status: 'present' | 'absent' | 'half_day' | 'leave') => {
    if (!user) return;

    try {
      // Check if attendance already exists for today
      const existing = attendance.find(a => a.employee_id === employeeId);

      if (existing) {
        const { error } = await supabase
          .from('attendance')
          .update({ status })
          .eq('id', existing.id);

        if (error) throw error;
      } else {
        const { error } = await supabase.from('attendance').insert({
          user_id: user.id,
          employee_id: employeeId,
          date: today,
          status,
          check_in_time: status === 'present' ? new Date().toISOString() : null,
        });

        if (error) throw error;
      }
      fetchData();
    } catch (err) {
      console.error('Error marking attendance:', err);
    }
  };

  const handleAddAdvance = async () => {
    if (!user || !selectedEmployee || !advanceForm.amount) return;

    try {
      const { error } = await supabase.from('salary_advances').insert({
        user_id: user.id,
        employee_id: selectedEmployee.id,
        amount: parseFloat(advanceForm.amount),
        date_given: today,
        reason: advanceForm.reason || null,
        is_deducted: false,
      });

      if (error) throw error;
      setShowAdvanceModal(false);
      setAdvanceForm({ amount: '', reason: '' });
      setSelectedEmployee(null);
      fetchData();
    } catch (err) {
      console.error('Error adding advance:', err);
      Alert.alert('Error', 'Failed to add advance');
    }
  };

  const getEmployeeAttendance = (employeeId: string) => {
    return attendance.find(a => a.employee_id === employeeId);
  };

  const getEmployeeAdvances = (employeeId: string) => {
    return advances.filter(a => a.employee_id === employeeId && !a.is_deducted);
  };

  const getTotalAdvances = (employeeId: string) => {
    return getEmployeeAdvances(employeeId).reduce((sum, a) => sum + a.amount, 0);
  };

  const getMonthAttendance = async (employeeId: string) => {
    if (!user) return { present: 0, absent: 0, halfDay: 0, leave: 0 };

    const startOfMonth = new Date(new Date().getFullYear(), new Date().getMonth(), 1).toISOString().split('T')[0];
    const { data } = await supabase
      .from('attendance')
      .select('status')
      .eq('user_id', user.id)
      .eq('employee_id', employeeId)
      .gte('date', startOfMonth);

    return {
      present: data?.filter(d => d.status === 'present').length || 0,
      absent: data?.filter(d => d.status === 'absent').length || 0,
      halfDay: data?.filter(d => d.status === 'half_day').length || 0,
      leave: data?.filter(d => d.status === 'leave').length || 0,
    };
  };

  const calculateSalary = (employee: Employee) => {
    const totalAdvance = getTotalAdvances(employee.id);
    const workingDays = 26; // Assume 26 working days
    const perDaySalary = employee.monthly_salary / workingDays;

    return {
      gross: employee.monthly_salary,
      advances: totalAdvance,
      net: employee.monthly_salary - totalAdvance,
      perDay: perDaySalary,
    };
  };

  const handleDeleteEmployee = async (id: string) => {
    Alert.alert(
      language === 'hi' ? 'हटाएं' : 'Delete',
      language === 'hi' ? 'क्या आप इस कर्मचारी को हटाना चाहते हैं?' : 'Delete this employee?',
      [
        { text: language === 'hi' ? 'नहीं' : 'No', style: 'cancel' },
        {
          text: language === 'hi' ? 'हां' : 'Yes',
          style: 'destructive',
          onPress: async () => {
            try {
              await supabase.from('employees').update({ is_active: false }).eq('id', id);
              fetchData();
            } catch (err) {
              console.error('Error:', err);
            }
          },
        },
      ]
    );
  };

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.header}>
        <TouchableOpacity style={styles.backBtn} onPress={() => router.back()}>
          <ArrowLeft color="#1e293b" size={24} strokeWidth={2} />
        </TouchableOpacity>
        <Text style={styles.headerTitle}>{language === 'hi' ? 'स्टाफ मैनेजमेंट' : 'Staff & Salary'}</Text>
        <TouchableOpacity style={styles.addBtn} onPress={() => setShowAddModal(true)}>
          <Plus color="#ffffff" size={22} strokeWidth={2.5} />
        </TouchableOpacity>
      </View>

      {/* Tabs */}
      <View style={styles.tabsContainer}>
        {[
          { key: 'staff', label: language === 'hi' ? 'स्टाफ' : 'Staff' },
          { key: 'attendance', label: language === 'hi' ? 'उपस्थिति' : 'Attendance' },
          { key: 'salary', label: language === 'hi' ? 'वेतन' : 'Salary' },
        ].map((tab) => (
          <TouchableOpacity
            key={tab.key}
            style={[styles.tab, activeTab === tab.key && styles.tabActive]}
            onPress={() => setActiveTab(tab.key as any)}
          >
            <Text style={[styles.tabText, activeTab === tab.key && styles.tabTextActive]}>{tab.label}</Text>
          </TouchableOpacity>
        ))}
      </View>

      <ScrollView
        style={styles.content}
        showsVerticalScrollIndicator={false}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} tintColor="#10b981" />}
      >
        {employees.length === 0 ? (
          <View style={styles.emptyState}>
            <Users color="#cbd5e1" size={48} />
            <Text style={styles.emptyText}>{language === 'hi' ? 'कोई कर्मचारी नहीं' : 'No employees yet'}</Text>
          </View>
        ) : (
          employees.map((employee) => {
            const isExpanded = expandedEmployee === employee.id;
            const todayAttendance = getEmployeeAttendance(employee.id);
            const salaryInfo = calculateSalary(employee);
            const employeeAdvances = getEmployeeAdvances(employee.id);

            return (
              <View key={employee.id} style={styles.employeeCard}>
                <TouchableOpacity
                  style={styles.employeeHeader}
                  onPress={() => setExpandedEmployee(isExpanded ? null : employee.id)}
                  activeOpacity={0.7}
                >
                  <View style={styles.employeeIcon}>
                    <Users color="#8b5cf6" size={22} />
                  </View>
                  <View style={styles.employeeInfo}>
                    <Text style={styles.employeeName}>{employee.name}</Text>
                    <View style={styles.employeeMeta}>
                      <Text style={styles.employeeRole}>{employee.role}</Text>
                      <Text style={styles.employeeDot}>•</Text>
                      <IndianRupee color="#94a3b8" size={12} />
                      <Text style={styles.employeeSalary}>{employee.monthly_salary.toLocaleString('en-IN')}/mo</Text>
                    </View>
                  </View>
                  {isExpanded ? <ChevronUp color="#64748b" size={20} /> : <ChevronDown color="#64748b" size={20} />}
                </TouchableOpacity>

                {/* Attendance Section */}
                {activeTab === 'attendance' && (
                  <View style={styles.attendanceSection}>
                    <Text style={styles.sectionLabel}>{language === 'hi' ? 'आज की उपस्थिति' : "Today's Attendance"}</Text>
                    <View style={styles.attendanceBtns}>
                      {[
                        { key: 'present', label: language === 'hi' ? 'हाजिर' : 'Present', color: '#10b981' },
                        { key: 'absent', label: language === 'hi' ? 'अनुपस्थित' : 'Absent', color: '#ef4444' },
                        { key: 'half_day', label: language === 'hi' ? 'आधा' : 'Half', color: '#f59e0b' },

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'नाम *' : 'Name *'}
              placeholderTextColor="#94a3b8"
              value={form.name}
              onChangeText={(text) => setForm({ ...form, name: text })}
            />

              style={styles.input}
              placeholder={language === 'hi' ? 'फोन नंबर' : 'Phone number'}

              keyboardType="phone-pad"
              value={form.phone}
              onChangeText={(text) => setForm({ ...form, phone: text })}
            />

              style={styles.input}
              placeholder={language === 'hi' ? 'पद' : 'Role'}
              placeholderTextColor="#94a3b8"
              value={form.role}

      </Modal>

      <Modal visible={showAdvanceModal} transparent animationType="slide" onRequestClose={() => setShowAdvanceModal(false)}>

          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>{language === 'hi' ? 'एडवांस/ऋण जोड़ें' : 'Add Salary Advance'}</Text>
            <Text style={styles.modalSubtitle}>{selectedEmployee?.name}</Text>

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'राशि *' : 'Amount *'}
              placeholderTextColor="#94a3b8"
              keyboardType="numeric"
              value={advanceForm.amount}
              onChangeText={(text) => setAdvanceForm({ ...advanceForm, amount: text })}
            />

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'कारण' : 'Reason'}
              placeholderTextColor="#94a3b8"
              value={advanceForm.reason}
              onChangeText={(text) => setAdvanceForm({ ...advanceForm, reason: text })}
            />

            <View style={styles.modalButtons}>
              <TouchableOpacity style={styles.cancelBtn} onPress={() => setShowAdvanceModal(false)}>
                <Text style={styles.cancelBtnText}>{language === 'hi' ? 'रद्द करें' : 'Cancel'}</Text>
              </TouchableOpacity>
              <TouchableOpacity style={styles.submitBtn} onPress={handleAddAdvance}>
                <Text style={styles.submitBtnText}>{language === 'hi' ? 'जोड़ें' : 'Add'}</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>
    </View>
  );
}

import { useState, useEffect } from 'react';
import { View, Text, StyleSheet, ScrollView, TouchableOpacity, Modal, TextInput, Alert, RefreshControl, Share } from 'react-native';
import { useRouter } from 'expo-router';
import { ArrowLeft, Plus, Truck, Phone, Mail, MapPin, FileText, Package, Send, ChevronDown, ChevronUp, Trash2, Edit, AlertTriangle } from 'lucide-react-native';
import { useAuth } from '@/contexts/AuthContext';
import { useI18n } from '@/contexts/I18nContext';
import { supabase } from '@/lib/supabase';

interface Supplier {
  id: string;
  name: string;
  phone: string | null;
  email: string | null;
  address: string | null;
  gstin: string | null;
  items_supplied: string[] | null;
}

interface LowStockItem {
  id: string;
  name: string;
  quantity: number;
  min_quantity: number;
  low_stock_threshold: number | null;
  supplier_id: string | null;
}

interface PurchaseOrder {
  id: string;
  supplier_id: string;
  items: { name: string; quantity: number }[];
  status: string;
  created_at: string;
  suppliers?: { name: string };
}

export default function SuppliersScreen() {
  const { user } = useAuth();
  const { language } = useI18n();
  const router = useRouter();
  const [suppliers, setSuppliers] = useState<Supplier[]>([]);
  const [lowStockItems, setLowStockItems] = useState<LowStockItem[]>([]);
  const [orders, setOrders] = useState<PurchaseOrder[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [showAddModal, setShowAddModal] = useState(false);
  const [showOrderModal, setShowOrderModal] = useState(false);
  const [selectedSupplier, setSelectedSupplier] = useState<Supplier | null>(null);
  const [orderItems, setOrderItems] = useState<{ name: string; quantity: string }[]>([]);
  const [expandedSupplier, setExpandedSupplier] = useState<string | null>(null);
  const [form, setForm] = useState({ name: '', phone: '', email: '', address: '', gstin: '', items: '' });

  useEffect(() => {
    fetchData();
  }, [user]);

  const fetchData = async () => {
    if (!user) return;
    setLoading(true);
    try {
      const [suppliersRes, productsRes, ordersRes] = await Promise.all([
        supabase.from('suppliers').select('*').eq('user_id', user.id).order('name'),
        supabase.from('products').select('id, name, quantity, min_quantity, low_stock_threshold, supplier_id').eq('user_id', user.id),
        supabase.from('purchase_orders').select('*, suppliers(name)').eq('user_id', user.id).order('created_at', { ascending: false }),
      ]);

      setSuppliers(suppliersRes.data || []);

      // Filter low stock items
      const lowStock = (productsRes.data || []).filter(p => {
        const threshold = p.low_stock_threshold || p.min_quantity || 5;
        return p.quantity <= threshold;
      });
      setLowStockItems(lowStock);
      setOrders(ordersRes.data || []);
    } catch (err) {
      console.error('Error fetching data:', err);
    } finally {
      setLoading(false);
    }
  };

  const onRefresh = async () => {
    setRefreshing(true);
    await fetchData();
    setRefreshing(false);
  };

  const handleAddSupplier = async () => {
    if (!user || !form.name) return;

    try {
      const itemsArray = form.items.split(',').map(i => i.trim()).filter(Boolean);
      const { error } = await supabase.from('suppliers').insert({
        user_id: user.id,
        name: form.name,
        phone: form.phone || null,
        email: form.email || null,
        address: form.address || null,
        gstin: form.gstin || null,
        items_supplied: itemsArray.length > 0 ? itemsArray : null,
      });

      if (error) throw error;
      setShowAddModal(false);
      setForm({ name: '', phone: '', email: '', address: '', gstin: '', items: '' });
      fetchData();
    } catch (err) {
      console.error('Error adding supplier:', err);
      Alert.alert('Error', 'Failed to add supplier');
    }
  };

  const handleGenerateOrder = (supplier: Supplier) => {
    setSelectedSupplier(supplier);
    // Auto-populate with low stock items from this supplier
    const supplierItems = lowStockItems.filter(i => i.supplier_id === supplier.id);
    if (supplierItems.length > 0) {
      setOrderItems(supplierItems.map(i => ({ name: i.name, quantity: String(i.min_quantity - i.quantity) || '10' })));
    } else {
      setOrderItems([{ name: '', quantity: '10' }]);
    }
    setShowOrderModal(true);
  };

  const handleCreateOrder = async () => {
    if (!user || !selectedSupplier || orderItems.length === 0) return;

    const validItems = orderItems.filter(i => i.name && i.quantity);
    if (validItems.length === 0) return;

    try {
      const { error } = await supabase.from('purchase_orders').insert({
        user_id: user.id,
        supplier_id: selectedSupplier.id,
        items: validItems.map(i => ({ name: i.name, quantity: parseInt(i.quantity) })),
        status: 'pending',
      });

      if (error) throw error;
      setShowOrderModal(false);
      setSelectedSupplier(null);
      setOrderItems([]);
      fetchData();
    } catch (err) {
      console.error('Error creating order:', err);
      Alert.alert('Error', 'Failed to create order');
    }
  };

  const shareOrderText = async (supplier: Supplier, items: { name: string; quantity: number }[]) => {
    const text = `*PURCHASE ORDER*\n\nSupplier: ${supplier.name}\nDate: ${new Date().toLocaleDateString('en-IN')}\n\nItems Required:\n${items.map(i => `- ${i.name}: ${i.quantity} units`).join('\n')}\n\nPlease confirm availability and total amount.\n\nSent from The Pocket CA App`;

    try {
      await Share.share({ message: text });
    } catch (err) {
      console.error('Error sharing:', err);
    }
  };

  const getSupplierLowStockCount = (supplierId: string) => {
    return lowStockItems.filter(i => i.supplier_id === supplierId).length;
  };

  const handleDeleteSupplier = async (id: string) => {
    Alert.alert(
      language === 'hi' ? 'हटाएं' : 'Delete',
      language === 'hi' ? 'क्या आप इस सप्लायर को हटाना चाहते हैं?' : 'Are you sure you want to delete this supplier?',
      [
        { text: language === 'hi' ? 'नहीं' : 'No', style: 'cancel' },
        {
          text: language === 'hi' ? 'हां' : 'Yes',
          style: 'destructive',
          onPress: async () => {
            try {
              await supabase.from('suppliers').delete().eq('id', id);
              fetchData();
            } catch (err) {
              console.error('Error deleting:', err);
            }
          },
        },
      ]
    );
  };

  return (
    <View style={styles.container}>
      {/* Header */}
      <View style={styles.header}>
        <TouchableOpacity style={styles.backBtn} onPress={() => router.back()}>
          <ArrowLeft color="#1e293b" size={24} strokeWidth={2} />
        </TouchableOpacity>
        <Text style={styles.headerTitle}>{language === 'hi' ? 'सप्लायर मैनेजमेंट' : 'Suppliers & Orders'}</Text>
        <TouchableOpacity style={styles.addBtn} onPress={() => setShowAddModal(true)}>
          <Plus color="#ffffff" size={22} strokeWidth={2.5} />
        </TouchableOpacity>
      </View>

      {/* Low Stock Alert */}
      {lowStockItems.length > 0 && (
        <View style={styles.alertBanner}>
          <AlertTriangle color="#f59e0b" size={20} />
          <Text style={styles.alertText}>
            {language === 'hi' ? `${lowStockItems.length} वस्तुएं कम स्टॉक में हैं` : `${lowStockItems.length} items low on stock`}
          </Text>
          <TouchableOpacity style={styles.alertBtn} onPress={() => {
            // Auto-generate order for all low stock
            if (suppliers.length > 0) {
              Alert.alert(language === 'hi' ? 'ऑर्डर बनाएं' : 'Create Order', language === 'hi' ? 'सप्लायर चुनें' : 'Select a supplier');
            }
          }}>
            <Text style={styles.alertBtnText}>{language === 'hi' ? 'ऑर्डर बनाएं' : 'Order'}</Text>
          </TouchableOpacity>
        </View>
      )}

      <ScrollView
        style={styles.content}
        showsVerticalScrollIndicator={false}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} tintColor="#10b981" />}
      >
        {/* Suppliers List */}
        <Text style={styles.sectionTitle}>{language === 'hi' ? 'सप्लायर सूची' : 'Suppliers'}</Text>

        {suppliers.length === 0 ? (
          <View style={styles.emptyState}>
            <Truck color="#cbd5e1" size={48} />
            <Text style={styles.emptyText}>{language === 'hi' ? 'कोई सप्लायर नहीं' : 'No suppliers yet'}</Text>
            <Text style={styles.emptyHint}>{language === 'hi' ? 'नया सप्लायर जोड़ें' : 'Add your first supplier'}</Text>
          </View>
        ) : (
          suppliers.map((supplier) => {
            const isExpanded = expandedSupplier === supplier.id;
            const lowStockCount = getSupplierLowStockCount(supplier.id);

            return (
              <View key={supplier.id} style={styles.supplierCard}>
                <TouchableOpacity
                  style={styles.supplierHeader}
                  onPress={() => setExpandedSupplier(isExpanded ? null : supplier.id)}
                  activeOpacity={0.7}
                >
                  <View style={styles.supplierIcon}>
                    <Truck color="#3b82f6" size={22} />
                    {lowStockCount > 0 && (
                      <View style={styles.badge}>
                        <Text style={styles.badgeText}>{lowStockCount}</Text>
                      </View>
                    )}
                  </View>
                  <View style={styles.supplierInfo}>
                    <Text style={styles.supplierName}>{supplier.name}</Text>
                    {supplier.phone && (
                      <View style={styles.supplierMeta}>
                        <Phone color="#94a3b8" size={12} />
                        <Text style={styles.supplierMetaText}>{supplier.phone}</Text>
                      </View>
                    )}
                  </View>
                  {isExpanded ? <ChevronUp color="#64748b" size={20} /> : <ChevronDown color="#64748b" size={20} />}
                </TouchableOpacity>

                {isExpanded && (
                  <View style={styles.supplierDetails}>
                    {supplier.email && (
                      <View style={styles.detailRow}>
                        <Mail color="#94a3b8" size={16} />
                        <Text style={styles.detailText}>{supplier.email}</Text>
                      </View>
                    )}
                    {supplier.address && (
                      <View style={styles.detailRow}>
                        <MapPin color="#94a3b8" size={16} />
                        <Text style={styles.detailText}>{supplier.address}</Text>
                      </View>
                    )}
                    {supplier.gstin && (
                      <View style={styles.detailRow}>
                        <FileText color="#94a3b8" size={16} />
                        <Text style={styles.detailText}>GSTIN: {supplier.gstin}</Text>
                      </View>
                    )}
                    {supplier.items_supplied && supplier.items_supplied.length > 0 && (
                      <View style={styles.itemsSupplied}>
                        <Text style={styles.itemsLabel}>{language === 'hi' ? 'आपूर्ति वस्तुएं:' : 'Items supplied:'}</Text>
                        <Text style={styles.itemsText}>{supplier.items_supplied.join(', ')}</Text>
                      </View>
                    )}
                    <View style={styles.actionRow}>
                      <TouchableOpacity style={styles.orderBtn} onPress={() => handleGenerateOrder(supplier)}>
                        <Send color="#ffffff" size={16} />
                        <Text style={styles.orderBtnText}>{language === 'hi' ? 'ऑर्डर बनाएं' : 'Create Order'}</Text>
                      </TouchableOpacity>
                      <TouchableOpacity style={styles.deleteBtn} onPress={() => handleDeleteSupplier(supplier.id)}>
                        <Trash2 color="#ef4444" size={18} />
                      </TouchableOpacity>
                    </View>
                  </View>
                )}
              </View>
            );
          })
        )}

        {/* Recent Orders */}
        {orders.length > 0 && (
          <>
            <Text style={styles.sectionTitle}>{language === 'hi' ? 'हाल के ऑर्डर' : 'Recent Orders'}</Text>
            {orders.slice(0, 5).map((order) => (
              <View key={order.id} style={styles.orderCard}>
                <View style={styles.orderHeader}>
                  <View style={styles.orderIcon}>
                    <Package color={order.status === 'pending' ? '#f59e0b' : '#10b981'} size={18} />
                  </View>
                  <View style={styles.orderInfo}>
                    <Text style={styles.orderSupplier}>{order.suppliers?.name || 'Unknown'}</Text>
                    <Text style={styles.orderDate}>
                      {new Date(order.created_at).toLocaleDateString('en-IN')}
                    </Text>
                  </View>
                  <View style={[styles.statusBadge, { backgroundColor: order.status === 'pending' ? 'rgba(245, 158, 11, 0.1)' : 'rgba(16, 185, 129, 0.1)' }]}>
                    <Text style={[styles.statusText, { color: order.status === 'pending' ? '#f59e0b' : '#10b981' }]}>
                      {order.status}
                    </Text>
                  </View>
                </View>
                <View style={styles.orderItems}>
                  {(order.items as any[])?.map((item, idx) => (
                    <Text key={idx} style={styles.orderItemText}>• {item.name}: {item.quantity}</Text>
                  ))}
                </View>
              </View>
            ))}
          </>
        )}

        <View style={styles.bottomSpace} />
      </ScrollView>

      {/* Add Supplier Modal */}
      <Modal visible={showAddModal} transparent animationType="slide" onRequestClose={() => setShowAddModal(false)}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>{language === 'hi' ? 'नया सप्लायर' : 'Add Supplier'}</Text>

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'नाम *' : 'Name *'}
              placeholderTextColor="#94a3b8"
              value={form.name}
              onChangeText={(text) => setForm({ ...form, name: text })}
            />

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'फोन नंबर' : 'Phone number'}
              placeholderTextColor="#94a3b8"
              keyboardType="phone-pad"
              value={form.phone}
              onChangeText={(text) => setForm({ ...form, phone: text })}
            />

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'ईमेल' : 'Email'}
              placeholderTextColor="#94a3b8"
              keyboardType="email-address"
              value={form.email}
              onChangeText={(text) => setForm({ ...form, email: text })}
            />

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'पता' : 'Address'}
              placeholderTextColor="#94a3b8"
              value={form.address}
              onChangeText={(text) => setForm({ ...form, address: text })}
            />

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'GSTIN' : 'GSTIN'}
              placeholderTextColor="#94a3b8"
              value={form.gstin}
              onChangeText={(text) => setForm({ ...form, gstin: text })}
            />

            <TextInput
              style={styles.input}
              placeholder={language === 'hi' ? 'आपूर्ति वस्तुएं (कॉमा से अलग)' : 'Items supplied (comma separated)'}
              placeholderTextColor="#94a3b8"
              value={form.items}
              onChangeText={(text) => setForm({ ...form, items: text })}

import { createContext, useContext, useEffect, useState, ReactNode } from 'react';
import { Session, User } from '@supabase/supabase-js';
import { supabase } from '@/lib/supabase';

interface AuthContextType {
  session: Session | null;
  user: User | null;
  loading: boolean;
  signIn: (email: string, password: string) => Promise<{ error: string | null }>;
  signUp: (email: string, password: string) => Promise<{ error: string | null }>;
  signOut: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      setLoading(false);
    });

    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
    });

    return () => subscription.unsubscribe();
  }, []);

  const signIn = async (email: string, password: string) => {
    const { error } = await supabase.auth.signInWithPassword({ email, password });
    return { error: error?.message || null };
  };

  const signUp = async (email: string, password: string) => {
    const { error } = await supabase.auth.signUp({ email, password });
    return { error: error?.message || null };
  };

  const signOut = async () => {
    await supabase.auth.signOut();
  };

  return (
    <AuthContext.Provider value={{
      session,
      user: session?.user ?? null,
      loading,
      signIn,
      signUp,
      signOut,
    }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (context === undefined) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
}

    setLanguage: (lang: Language) => void;
    t: (key: string) => string;
    toggleLanguage: () => void;
  }
  
  const I18nContext = createContext<I18nContextType | undefined>(undefined);
  
  export function I18nProvider({ children }: { children: ReactNode }) {
    const [language, setLanguageState] = useState<Language>('en');
  
    const setLanguage = (lang: Language) => {
      setLanguageState(lang);
    };
  
    const t = (key: string): string => {
      const translation = translations[key];
      if (!translation) return key;
      return translation[language] || translation.en || key;
    };
  
    const toggleLanguage = () => {
      setLanguageState(prev => prev === 'en' ? 'hi' : 'en');
    };
  
    const value = useMemo(() => ({
      language,
      setLanguage,
      t,
      toggleLanguage,
    }), [language]);
  
    return (
      <I18nContext.Provider value={value}>
        {children}
      </I18nContext.Provider>
    );
  }
  
  export function useI18n() {
    const context = useContext(I18nContext);
    if (context === undefined) {
      throw new Error('useI18n must be used within an I18nProvider');
    }
    return context;
  }

## Purpose
Accounting app for shopkeepers with multi-user support. Each user has their own sales, expenses, and ledger entries.

## New Tables
1. `sales` - Income/sales records
   - id (uuid, primary key)
   - description (text, what was sold)
   - amount (decimal, sale amount before tax)
   - gst (decimal, GST amount)
   - total (decimal, amount + gst)
   - date (date, when sale occurred)
   - user_id (uuid, owner, defaults to auth.uid())
   - created_at (timestamp)

2. `expenses` - Expense/cost records
   - id (uuid, primary key)
   - description (text, what was purchased)
   - amount (decimal, expense amount before tax)
   - gst (decimal, GST amount)
   - total (decimal, amount + gst)
   - category (text, expense category)
   - date (date, when expense occurred)
   - user_id (uuid, owner, defaults to auth.uid())
   - created_at (timestamp)

3. `ledger_entries` - Scanned bill/invoice items
   - id (uuid, primary key)
   - item_name (text, item description)
   - price (decimal, base price)
   - gst (decimal, GST amount)
   - total (decimal, price + gst)
   - category (text, sale or expense)
   - date_scanned (date, when scanned)
   - user_id (uuid, owner, defaults to auth.uid())
   - created_at (timestamp)

## Security
- RLS enabled on all tables
- Owner-scoped policies: users can only access their own data
- All tables have user_id DEFAULT auth.uid() for seamless inserts
*/

-- Sales table
CREATE TABLE IF NOT EXISTS sales (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  description text NOT NULL,
  amount decimal(12,2) NOT NULL,
  gst decimal(12,2) NOT NULL DEFAULT 0,
  total decimal(12,2) NOT NULL,
  date date NOT NULL DEFAULT CURRENT_DATE,
  user_id uuid NOT NULL DEFAULT auth.uid() REFERENCES auth.users(id) ON DELETE CASCADE,
  created_at timestamptz DEFAULT now()
);

  category text,
  date date NOT NULL DEFAULT CURRENT_DATE,
  user_id uuid NOT NULL DEFAULT auth.uid() REFERENCES auth.users(id) ON DELETE CASCADE,

);

  total decimal(12,2) NOT NULL,
  category text NOT NULL DEFAULT 'expense',
  date_scanned date NOT NULL DEFAULT CURRENT_DATE,
  user_id uuid NOT NULL DEFAULT auth.uid() REFERENCES auth.users(id) ON DELETE CASCADE,
  created_at timestamptz DEFAULT now()
